Calling Methods

Build a generic call method with request ID tracking.

A Generic Call Method

Instead of writing separate code for each MCP method, we can create a generic call method that handles all the common logic: ID generation, request formatting, and response parsing.

python 
from typing import Any, Optional

class MCPClientError(Exception):
    """Base exception for MCP client errors."""
    pass

class MCPProtocolError(MCPClientError):
    """Error in the MCP protocol (JSON-RPC error from server)."""
    def __init__(self, code: int, message: str, data: Any = None):
        self.code = code
        self.message = message
        self.data = data
        super().__init__(f"MCP Error {code}: {message}")


def call(self, method: str, params: Optional[dict] = None) -> Any:
    """
    Call an MCP method and return the result.

    Args:
        method: The MCP method name (e.g., "tools/list")
        params: Optional parameters for the method

    Returns:
        The result from the server response

    Raises:
        MCPProtocolError: If the server returns an error
        RuntimeError: If not connected or initialized
    """
    if not self.initialized:
        raise RuntimeError("Client not initialized. Call initialize() first.")

    # Send request and get response
    response = self.send_request(method, params)

    # Check for error response
    if "error" in response:
        error = response["error"]
        raise MCPProtocolError(
            code=error.get("code", -1),
            message=error.get("message", "Unknown error"),
            data=error.get("data")
        )

    # Return the result
    return response.get("result")

Request ID Management

Each request needs a unique ID to match with its response. Here's a robust ID tracker:

python 
import threading
from typing import Dict, Optional
from dataclasses import dataclass, field
from datetime import datetime

@dataclass
class PendingRequest:
    """Track a pending request."""
    id: int
    method: str
    sent_at: datetime = field(default_factory=datetime.now)
    params: Optional[dict] = None


class RequestTracker:
    """Thread-safe request ID tracker."""

    def __init__(self):
        self._next_id = 1
        self._pending: Dict[int, PendingRequest] = {}
        self._lock = threading.Lock()

    def next_id(self) -> int:
        """Get the next request ID."""
        with self._lock:
            request_id = self._next_id
            self._next_id += 1
            return request_id

    def track(self, request_id: int, method: str, params: Optional[dict] = None):
        """Track a pending request."""
        with self._lock:
            self._pending[request_id] = PendingRequest(
                id=request_id,
                method=method,
                params=params
            )

    def resolve(self, request_id: int) -> Optional[PendingRequest]:
        """Mark a request as resolved and return its info."""
        with self._lock:
            return self._pending.pop(request_id, None)

    def pending_count(self) -> int:
        """Return the number of pending requests."""
        with self._lock:
            return len(self._pending)

Adding Timeouts

Requests should timeout if the server doesn't respond:

python 
import select
import time

class TimeoutError(MCPClientError):
    """Request timed out."""
    pass


def call_with_timeout(self, method: str, params: Optional[dict] = None,
                      timeout: float = 30.0) -> Any:
    """
    Call an MCP method with a timeout.

    Args:
        method: The MCP method name
        params: Optional parameters
        timeout: Timeout in seconds (default 30)

    Returns:
        The result from the server

    Raises:
        TimeoutError: If the request times out
    """
    if not self.initialized:
        raise RuntimeError("Client not initialized")

    # Create and send request
    request_id = self.tracker.next_id()
    request = {
        "jsonrpc": "2.0",
        "id": request_id,
        "method": method,
    }
    if params:
        request["params"] = params

    self.tracker.track(request_id, method, params)

    # Send request
    message = json.dumps(request) + "\n"
    self.process.stdin.write(message)
    self.process.stdin.flush()

    # Wait for response with timeout
    start_time = time.time()
    while True:
        # Check timeout
        elapsed = time.time() - start_time
        if elapsed >= timeout:
            self.tracker.resolve(request_id)
            raise TimeoutError(f"Request {method} timed out after {timeout}s")

        # Check if data available (Unix only - use threads for Windows)
        remaining = timeout - elapsed
        readable, _, _ = select.select(
            [self.process.stdout], [], [], min(remaining, 0.1)
        )

        if readable:
            line = self.process.stdout.readline()
            if line:
                response = json.loads(line.strip())
                if response.get("id") == request_id:
                    self.tracker.resolve(request_id)
                    if "error" in response:
                        error = response["error"]
                        raise MCPProtocolError(
                            error.get("code", -1),
                            error.get("message", "Unknown error"),
                            error.get("data")
                        )
                    return response.get("result")
Platform Considerations

The select module works differently on Windows. For cross-platform code, consider using threads or async I/O.

High-Level Method Wrappers

With a generic call method, you can create convenient wrappers for specific MCP methods:

python 
def call_tool(self, name: str, arguments: Optional[dict] = None) -> Any:
    """
    Call a tool on the server.

    Args:
        name: The tool name
        arguments: Arguments to pass to the tool

    Returns:
        The tool's result content
    """
    result = self.call("tools/call", {
        "name": name,
        "arguments": arguments or {}
    })
    return result.get("content", [])


def read_resource(self, uri: str) -> Any:
    """
    Read a resource from the server.

    Args:
        uri: The resource URI

    Returns:
        The resource contents
    """
    result = self.call("resources/read", {"uri": uri})
    return result.get("contents", [])


def get_prompt(self, name: str, arguments: Optional[dict] = None) -> Any:
    """
    Get a prompt from the server.

    Args:
        name: The prompt name
        arguments: Arguments for the prompt

    Returns:
        The prompt messages
    """
    result = self.call("prompts/get", {
        "name": name,
        "arguments": arguments or {}
    })
    return result.get("messages", [])

Error Handling Best Practices

Handle different types of errors appropriately:

python 
try:
    result = client.call_tool("get_weather", {"city": "London"})
    print(f"Weather: {result}")

except MCPProtocolError as e:
    if e.code == -32601:
        print("Method not found - server doesn't support this")
    elif e.code == -32602:
        print(f"Invalid params: {e.message}")
    else:
        print(f"Server error: {e.message}")

except TimeoutError:
    print("Request timed out - server may be overloaded")

except RuntimeError as e:
    print(f"Connection error: {e}")
Standard JSON-RPC Error Codes

-32700: Parse error
-32600: Invalid request
-32601: Method not found
-32602: Invalid params
-32603: Internal error
