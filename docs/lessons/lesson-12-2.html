Planning Your Server

Design the architecture and data model for your MCP server.

Planning Before Coding

Before writing any code, let us plan our server architecture. Good planning helps avoid rewrites and makes implementation smoother. We will define:

  • The data model for notes
  • The tool interfaces and their parameters
  • The resource URIs and their responses
  • The prompt templates and their arguments

Data Model

Each note in our system will have the following structure:

python 
from dataclasses import dataclass, field
from typing import List, Optional
from datetime import datetime
import uuid


@dataclass
class Note:
    """Represents a single note."""
    id: str
    title: str
    content: str
    tags: List[str] = field(default_factory=list)
    created_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    updated_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())

    def to_dict(self) -> dict:
        """Convert to dictionary for JSON serialization."""
        return {
            "id": self.id,
            "title": self.title,
            "content": self.content,
            "tags": self.tags,
            "created_at": self.created_at,
            "updated_at": self.updated_at,
        }

    @classmethod
    def from_dict(cls, data: dict) -> "Note":
        """Create a Note from a dictionary."""
        return cls(
            id=data["id"],
            title=data["title"],
            content=data["content"],
            tags=data.get("tags", []),
            created_at=data.get("created_at", datetime.utcnow().isoformat()),
            updated_at=data.get("updated_at", datetime.utcnow().isoformat()),
        )

    @staticmethod
    def generate_id() -> str:
        """Generate a unique ID for a new note."""
        return str(uuid.uuid4())[:8]

File Storage

We will use a simple JSON file for persistence. Here is the storage helper:

python 
import json
import os
from typing import List, Optional


class NotesStorage:
    """Handle persistent storage of notes."""

    def __init__(self, filepath: str = "notes.json"):
        self.filepath = filepath
        self.notes: List[Note] = []
        self.load()

    def load(self):
        """Load notes from file."""
        if os.path.exists(self.filepath):
            try:
                with open(self.filepath, "r") as f:
                    data = json.load(f)
                    self.notes = [Note.from_dict(n) for n in data]
            except (json.JSONDecodeError, KeyError):
                self.notes = []
        else:
            self.notes = []

    def save(self):
        """Save notes to file."""
        with open(self.filepath, "w") as f:
            json.dump([n.to_dict() for n in self.notes], f, indent=2)

    def get_by_id(self, note_id: str) -> Optional[Note]:
        """Find a note by ID."""
        for note in self.notes:
            if note.id == note_id:
                return note
        return None

    def get_all_tags(self) -> List[str]:
        """Get all unique tags across notes."""
        tags = set()
        for note in self.notes:
            tags.update(note.tags)
        return sorted(list(tags))
Production Considerations

For a production server, you would want to use a proper database, add file locking for concurrent access, and implement backup/recovery. For this capstone, JSON file storage is sufficient to demonstrate the MCP concepts.
