Building Calculator Tools

Put it all together by implementing add, subtract, and multiply tools.

Building Incrementally

Let's build a complete MCP calculator server step by step. We'll create calculator_server.py incrementally, testing as we go.

File Structure

We'll build one file: calculator_server.py. Each section below adds to this file. Follow along and add each section in order.

Section 1: Imports and Tool Definitions

Create: calculator_server.py

Start with the imports and tool definitions:

python calculator_server.py 
"""
calculator_server.py - MCP server with calculator tools.

Run with: python calculator_server.py
Test with: curl commands
"""

import json
from http.server import HTTPServer, BaseHTTPRequestHandler
from typing import cast


# =============================================================================
# Section 1: Tool Definitions
# =============================================================================

TOOLS = [
    {
        "name": "add",
        "description": "Add two numbers together and return the sum",
        "inputSchema": {
            "type": "object",
            "properties": {
                "a": {
                    "type": "number",
                    "description": "First number to add"
                },
                "b": {
                    "type": "number",
                    "description": "Second number to add"
                }
            },
            "required": ["a", "b"]
        }
    },
    {
        "name": "subtract",
        "description": "Subtract the second number from the first and return the difference",
        "inputSchema": {
            "type": "object",
            "properties": {
                "a": {
                    "type": "number",
                    "description": "Number to subtract from"
                },
                "b": {
                    "type": "number",
                    "description": "Number to subtract"
                }
            },
            "required": ["a", "b"]
        }
    },
    {
        "name": "multiply",
        "description": "Multiply two numbers and return the product",
        "inputSchema": {
            "type": "object",
            "properties": {
                "a": {
                    "type": "number",
                    "description": "First number to multiply"
                },
                "b": {
                    "type": "number",
                    "description": "Second number to multiply"
                }
            },
            "required": ["a", "b"]
        }
    }
]


# Quick test - print tool names
if __name__ == "__main__":
    print("Defined tools:")
    for tool in TOOLS:
        print(f"  - {tool['name']}: {tool['description']}")

# =============================================================================
# Section 2: Result Helpers
# =============================================================================

def text_result(text: str) -> dict:
    """Create a successful text result."""
    return {
        "content": [{"type": "text", "text": str(text)}]
    }


def error_result(message: str) -> dict:
    """Create an error result."""
    return {
        "content": [{"type": "text", "text": f"Error: {message}"}],
        "isError": True
    }

Section 3: Tool Implementations

Edit: calculator_server.py

Add the following after Section 2:

python 
# =============================================================================
# Section 3: Tool Implementations
# =============================================================================

def execute_add(arguments: dict) -> dict:
    """Add two numbers."""
    a = arguments.get("a")
    b = arguments.get("b")

    if a is None or b is None:
        return error_result("Both 'a' and 'b' arguments are required")

    if not isinstance(a, (int, float)) or not isinstance(b, (int, float)):
        return error_result("Arguments must be numbers")

    result = a + b
    return text_result(f"{a} + {b} = {result}")


def execute_subtract(arguments: dict) -> dict:
    """Subtract two numbers."""
    a = arguments.get("a")
    b = arguments.get("b")

    if a is None or b is None:
        return error_result("Both 'a' and 'b' arguments are required")

    if not isinstance(a, (int, float)) or not isinstance(b, (int, float)):
        return error_result("Arguments must be numbers")

    result = a - b
    return text_result(f"{a} - {b} = {result}")


def execute_multiply(arguments: dict) -> dict:
    """Multiply two numbers."""
    a = arguments.get("a")
    b = arguments.get("b")

    if a is None or b is None:
        return error_result("Both 'a' and 'b' arguments are required")

    if not isinstance(a, (int, float)) or not isinstance(b, (int, float)):
        return error_result("Arguments must be numbers")

    result = a * b
    return text_result(f"{a} * {b} = {result}")


# Map tool names to their handler functions
TOOL_HANDLERS = {
    "add": execute_add,
    "subtract": execute_subtract,
    "multiply": execute_multiply,
}

Section 4: MCP Method Handlers

Edit: calculator_server.py

Add the following after Section 3:

python 
# =============================================================================
# Section 4: MCP Method Handlers
# =============================================================================

def handle_initialize(request_id: int | str | None, params: dict) -> dict:
    """Handle the initialize method."""
    return {
        "jsonrpc": "2.0",
        "id": request_id,
        "result": {
            "protocolVersion": "2024-11-05",
            "capabilities": {
                "tools": {}  # We support tools
            },
            "serverInfo": {
                "name": "calculator-server",
                "version": "1.0.0"
            }
        }
    }


def handle_tools_list(request_id: int | str | None, params: dict) -> dict:
    """Handle the tools/list method."""
    return {
        "jsonrpc": "2.0",
        "id": request_id,
        "result": {
            "tools": TOOLS
        }
    }


def handle_tools_call(request_id: int | str | None, params: dict) -> dict:
    """Handle the tools/call method."""
    tool_name = params.get("name")
    arguments = params.get("arguments", {})

    # Check if tool exists
    if tool_name not in TOOL_HANDLERS:
        return {
            "jsonrpc": "2.0",
            "id": request_id,
            "error": {
                "code": -32602,
                "message": f"Unknown tool: {tool_name}"
            }
        }

    # Execute the tool
    handler = TOOL_HANDLERS[tool_name]
    result = handler(arguments)

    return {
        "jsonrpc": "2.0",
        "id": request_id,
        "result": result
    }

Section 5: Request Router

Edit: calculator_server.py

Add the following after Section 4:

python 
# =============================================================================
# Section 5: Request Router
# =============================================================================

def handle_request(request: dict) -> dict | None:
    """Route JSON-RPC requests to appropriate handlers."""
    method = request.get("method")
    request_id = cast(int | str | None, request.get("id"))
    params = request.get("params", {})

    # Route to handler based on method
    handlers = {
        "initialize": handle_initialize,
        "tools/list": handle_tools_list,
        "tools/call": handle_tools_call,
    }

    handler = handlers.get(method)
    if handler:
        return handler(request_id, params)

    # Handle notifications (no response needed)
    if method == "notifications/initialized":
        return None

    # Method not found
    return {
        "jsonrpc": "2.0",
        "id": request_id,
        "error": {
            "code": -32601,
            "message": f"Method not found: {method}"
        }
    }

Section 6: HTTP Server

Edit: calculator_server.py

Add the following after Section 5:

python 
# =============================================================================
# Section 6: HTTP Server
# =============================================================================

class MCPHandler(BaseHTTPRequestHandler):
    """HTTP handler for MCP requests."""

    def log_message(self, format, *args):
        """Custom logging."""
        print(f"[MCP] {args[0]}")

    def send_json(self, data: dict, status: int = 200):
        """Send a JSON response."""
        body = json.dumps(data, indent=2).encode("utf-8")
        self.send_response(status)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def do_POST(self):
        """Handle POST requests."""
        # Read and parse JSON body
        content_length = int(self.headers.get("Content-Length", 0))
        body = self.rfile.read(content_length)

        try:
            request = json.loads(body.decode("utf-8"))
        except json.JSONDecodeError as e:
            self.send_json({
                "jsonrpc": "2.0",
                "id": None,
                "error": {"code": -32700, "message": f"Parse error: {e}"}
            }, status=400)
            return

        # Validate JSON-RPC structure
        if request.get("jsonrpc") != "2.0":
            self.send_json({
                "jsonrpc": "2.0",
                "id": request.get("id"),
                "error": {"code": -32600, "message": "Invalid Request: jsonrpc must be '2.0'"}
            }, status=400)
            return

        # Handle the request
        response = handle_request(request)

        # Notifications don't get responses
        if response is None:
            self.send_response(204)
            self.end_headers()
            return

        self.send_json(response)


def run_server(host: str = "localhost", port: int = 8080):
    """Start the MCP calculator server."""
    server = HTTPServer((host, port), MCPHandler)
    print(f"Calculator MCP Server running on http://{host}:{port}")
    print("Available tools: add, subtract, multiply")
    print("Press Ctrl+C to stop")

    try:
        server.serve_forever()
    except KeyboardInterrupt:
        print("\nShutting down...")
        server.shutdown()

What You've Built

Congratulations! Your calculator server now:

  • Defines three tools with proper inputSchema
  • Handles tools/list to advertise capabilities
  • Handles tools/call to execute tools
  • Returns properly formatted results with content arrays
  • Handles errors gracefully with isError flag
  • Validates arguments before execution

Try adding more tools: divide (with division-by-zero handling), power, sqrt, or even evaluate for complex expressions!
