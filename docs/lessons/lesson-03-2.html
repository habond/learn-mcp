Request & Response

Learn to create valid JSON-RPC requests and match responses by ID.

Anatomy of a Request

A JSON-RPC 2.0 request has four fields. Let's examine each one in detail and understand how to construct valid requests in Python.

python 
def create_request(method: str, params: dict = None, request_id: int = None) -> dict:
    """
    Create a valid JSON-RPC 2.0 request.

    Args:
        method: The method name to call (e.g., "tools/list")
        params: Optional parameters for the method
        request_id: Unique identifier for matching responses

    Returns:
        A properly formatted JSON-RPC request dict
    """
    request = {
        "jsonrpc": "2.0",  # Always exactly "2.0"
        "method": method,   # Required: the method to call
    }

    # ID is required for requests (not notifications)
    if request_id is not None:
        request["id"] = request_id

    # Params are optional
    if params is not None:
        request["params"] = params

    return request


# Example usage
request = create_request(
    method="tools/call",
    params={"name": "calculator", "arguments": {"expression": "2 + 2"}},
    request_id=1
)

print(request)
# {
#     "jsonrpc": "2.0",
#     "id": 1,
#     "method": "tools/call",
#     "params": {"name": "calculator", "arguments": {"expression": "2 + 2"}}
# }

Generating Request IDs

Request IDs must be unique within a session. Here's a robust way to generate them:

python 
import threading


class RequestIdGenerator:
    """Thread-safe request ID generator."""

    def __init__(self):
        self._counter = 0
        self._lock = threading.Lock()

    def next_id(self) -> int:
        """Generate the next unique request ID."""
        with self._lock:
            self._counter += 1
            return self._counter


# Usage
id_generator = RequestIdGenerator()

request1 = create_request("tools/list", request_id=id_generator.next_id())
request2 = create_request("resources/list", request_id=id_generator.next_id())

print(request1["id"])  # 1
print(request2["id"])  # 2
ID Types

JSON-RPC allows IDs to be strings, integers, or null. MCP conventionally uses incrementing integers, but UUIDs or other string identifiers also work.

Building Success Responses

When a request succeeds, the server responds with a result field. The id must match the request's ID exactly.

python 
def create_success_response(request_id, result: dict) -> dict:
    """
    Create a success response for a JSON-RPC request.

    Args:
        request_id: The ID from the original request (must match exactly)
        result: The result data to return

    Returns:
        A properly formatted JSON-RPC success response
    """
    return {
        "jsonrpc": "2.0",
        "id": request_id,  # Must match the request ID
        "result": result
    }


# Example: responding to a tools/list request
response = create_success_response(
    request_id=1,
    result={
        "tools": [
            {
                "name": "calculator",
                "description": "Performs math calculations",
                "inputSchema": {
                    "type": "object",
                    "properties": {
                        "expression": {"type": "string"}
                    }
                }
            }
        ]
    }
)

print(response)
# {
#     "jsonrpc": "2.0",
#     "id": 1,
#     "result": {
#         "tools": [...]
#     }
# }

Matching Responses to Requests

In asynchronous systems, responses may arrive out of order. Use a pending request registry to match responses back to their original requests.

python 
import asyncio
from typing import Any


class PendingRequests:
    """Track pending requests and match them to responses."""

    def __init__(self):
        self._pending: dict[int, asyncio.Future] = {}
        self._id_generator = RequestIdGenerator()

    def create_request(self, method: str, params: dict = None) -> tuple[dict, asyncio.Future]:
        """
        Create a request and register it as pending.

        Returns:
            A tuple of (request_dict, future_for_response)
        """
        request_id = self._id_generator.next_id()

        # Create a future to hold the eventual response
        future = asyncio.get_event_loop().create_future()
        self._pending[request_id] = future

        request = create_request(method, params, request_id)
        return request, future

    def resolve(self, response: dict) -> bool:
        """
        Match a response to its pending request.

        Returns:
            True if the response was matched, False otherwise
        """
        request_id = response.get("id")

        if request_id not in self._pending:
            return False  # Unknown ID - possibly a late response

        future = self._pending.pop(request_id)

        # Check if it's an error response
        if "error" in response:
            future.set_exception(
                JsonRpcError(response["error"]["code"], response["error"]["message"])
            )
        else:
            future.set_result(response.get("result"))

        return True


class JsonRpcError(Exception):
    """Represents a JSON-RPC error response."""

    def __init__(self, code: int, message: str, data=None):
        self.code = code
        self.message = message
        self.data = data
        super().__init__(f"JSON-RPC Error {code}: {message}")

Complete Request/Response Flow

Here's a complete example showing how requests and responses flow through a client-server interaction:

python 
import json


class JsonRpcClient:
    """A simple JSON-RPC client demonstrating request/response matching."""

    def __init__(self):
        self._pending = PendingRequests()

    async def call(self, method: str, params: dict = None) -> Any:
        """
        Make a JSON-RPC call and wait for the response.

        Args:
            method: The method to call
            params: Optional parameters

        Returns:
            The result from the server

        Raises:
            JsonRpcError: If the server returns an error
        """
        # Create request and get future for response
        request, future = self._pending.create_request(method, params)

        # Send the request (in real code, this would go over HTTP/stdio)
        print(f"Sending: {json.dumps(request)}")
        await self._send(request)

        # Wait for the response
        result = await future
        return result

    async def _send(self, request: dict):
        """Send request to server. Override in subclass."""
        pass

    def handle_response(self, response: dict):
        """Process an incoming response from the server."""
        if not self._pending.resolve(response):
            print(f"Warning: Received response for unknown ID: {response.get('id')}")


# Example usage (pseudocode)
async def example():
    client = JsonRpcClient()

    # Make concurrent requests
    tools_future = client.call("tools/list")
    resources_future = client.call("resources/list")

    # Both requests sent, now wait for responses
    # They can arrive in any order!
    tools = await tools_future
    resources = await resources_future

    print(f"Got {len(tools['tools'])} tools")
    print(f"Got {len(resources['resources'])} resources")
Why IDs Matter

Without IDs, concurrent requests would be impossible. The client wouldn't know which response belongs to which request. IDs enable:

  • Concurrent (parallel) requests
  • Out-of-order response handling
  • Request timeouts and cancellation
  • Error attribution to specific requests

Validation Rules

Always validate incoming requests and responses. Here's a validation function:

python 
def validate_request(request: dict) -> tuple[bool, str]:
    """
    Validate a JSON-RPC request.

    Returns:
        A tuple of (is_valid, error_message)
    """
    if not isinstance(request, dict):
        return False, "Request must be a JSON object"

    if request.get("jsonrpc") != "2.0":
        return False, "jsonrpc field must be exactly '2.0'"

    if "method" not in request:
        return False, "Missing required 'method' field"

    if not isinstance(request["method"], str):
        return False, "method must be a string"

    # ID validation (required for requests, absent for notifications)
    if "id" in request:
        id_val = request["id"]
        if not isinstance(id_val, (str, int)) or id_val is None:
            return False, "id must be a string or integer, not null"

    # Params must be object or array if present
    if "params" in request:
        params = request["params"]
        if not isinstance(params, (dict, list)):
            return False, "params must be an object or array"

    return True, ""


def validate_response(response: dict) -> tuple[bool, str]:
    """
    Validate a JSON-RPC response.

    Returns:
        A tuple of (is_valid, error_message)
    """
    if not isinstance(response, dict):
        return False, "Response must be a JSON object"

    if response.get("jsonrpc") != "2.0":
        return False, "jsonrpc field must be exactly '2.0'"

    if "id" not in response:
        return False, "Response must have an 'id' field"

    # Must have either result or error, not both
    has_result = "result" in response
    has_error = "error" in response

    if has_result and has_error:
        return False, "Response cannot have both 'result' and 'error'"

    if not has_result and not has_error:
        return False, "Response must have either 'result' or 'error'"

    return True, ""
