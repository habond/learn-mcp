Listing Primitives

Implement list_tools(), list_resources(), and list_prompts() methods.

Discovering Server Capabilities

After initialization, you'll want to discover what the server offers. MCP provides three "list" methods for the three primitive types:

Method Returns Use For
tools/list List of tools Functions the server can execute
resources/list List of resources Data the server can provide
prompts/list List of prompts Templates for AI interactions

Listing Tools

The tools/list method returns all available tools:

python 
from dataclasses import dataclass
from typing import List, Optional, Any

@dataclass
class Tool:
    """Represents an MCP tool."""
    name: str
    description: str
    input_schema: dict

    @classmethod
    def from_dict(cls, data: dict) -> "Tool":
        return cls(
            name=data["name"],
            description=data.get("description", ""),
            input_schema=data.get("inputSchema", {})
        )


def list_tools(self) -> List[Tool]:
    """
    List all tools available on the server.

    Returns:
        List of Tool objects
    """
    if not self.has_tools():
        return []

    result = self.call("tools/list")
    tools_data = result.get("tools", [])

    return [Tool.from_dict(t) for t in tools_data]

Listing Resources

Resources are data that the server can provide:

python 
@dataclass
class Resource:
    """Represents an MCP resource."""
    uri: str
    name: str
    description: str
    mime_type: Optional[str] = None

    @classmethod
    def from_dict(cls, data: dict) -> "Resource":
        return cls(
            uri=data["uri"],
            name=data.get("name", data["uri"]),
            description=data.get("description", ""),
            mime_type=data.get("mimeType")
        )


def list_resources(self) -> List[Resource]:
    """
    List all resources available on the server.

    Returns:
        List of Resource objects
    """
    if not self.has_resources():
        return []

    result = self.call("resources/list")
    resources_data = result.get("resources", [])

    return [Resource.from_dict(r) for r in resources_data]

Listing Prompts

Prompts are templates that can be filled with arguments:

python 
@dataclass
class PromptArgument:
    """Represents a prompt argument."""
    name: str
    description: str
    required: bool = False

    @classmethod
    def from_dict(cls, data: dict) -> "PromptArgument":
        return cls(
            name=data["name"],
            description=data.get("description", ""),
            required=data.get("required", False)
        )


@dataclass
class Prompt:
    """Represents an MCP prompt."""
    name: str
    description: str
    arguments: List[PromptArgument]

    @classmethod
    def from_dict(cls, data: dict) -> "Prompt":
        args = data.get("arguments", [])
        return cls(
            name=data["name"],
            description=data.get("description", ""),
            arguments=[PromptArgument.from_dict(a) for a in args]
        )


def list_prompts(self) -> List[Prompt]:
    """
    List all prompts available on the server.

    Returns:
        List of Prompt objects
    """
    if not self.has_prompts():
        return []

    result = self.call("prompts/list")
    prompts_data = result.get("prompts", [])

    return [Prompt.from_dict(p) for p in prompts_data]

Displaying Server Inventory

Here's a utility function to display everything a server offers:

python 
def print_server_inventory(client):
    """Print a summary of what the server offers."""
    print(f"\n{'='*60}")
    print(f"Server: {client.server_info.get('name', 'Unknown')}")
    print(f"Version: {client.server_info.get('version', 'Unknown')}")
    print(f"{'='*60}\n")

    # Tools
    if client.has_tools():
        tools = client.list_tools()
        print(f"TOOLS ({len(tools)}):")
        for tool in tools:
            print(f"  - {tool.name}: {tool.description}")
        print()
    else:
        print("TOOLS: None\n")

    # Resources
    if client.has_resources():
        resources = client.list_resources()
        print(f"RESOURCES ({len(resources)}):")
        for resource in resources:
            print(f"  - {resource.uri}")
            print(f"    {resource.description}")
        print()
    else:
        print("RESOURCES: None\n")

    # Prompts
    if client.has_prompts():
        prompts = client.list_prompts()
        print(f"PROMPTS ({len(prompts)}):")
        for prompt in prompts:
            args = ", ".join(a.name for a in prompt.arguments)
            print(f"  - {prompt.name}({args})")
            print(f"    {prompt.description}")
        print()
    else:
        print("PROMPTS: None\n")

Caching Lists

For performance, consider caching the results of list methods. The server will send a notifications/tools/list_changed notification if the list changes (if it supports that capability).
