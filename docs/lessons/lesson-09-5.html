Building Useful Prompts

Create practical prompts for code review, summarization, and explanation tasks.

Building a Complete Prompts Server

Let's build a complete MCP server with useful prompts for real-world tasks. We'll create this incrementally in 5 sections.

What We're Building

A prompts server with four practical prompts: code review, text summarization, code explanation, and git commit message generation.

Section 1: Imports and Prompt Definitions

Start by creating the file with imports and prompt metadata.

Create: prompts_server.py

python prompts_server.py 
#!/usr/bin/env python3
"""
prompts_server.py - MCP server with useful prompt templates.

This server provides prompts for:
- Code review with configurable focus areas
- Text summarization in various styles
- Code explanation for different audiences

from http.server import HTTPServer, BaseHTTPRequestHandler
from typing import Any, cast

Add the functions that generate prompt content based on arguments.

python 
# =============================================================================
# Prompt Generators
# =============================================================================

def generate_code_review(arguments: dict) -> dict:
    """Generate a thorough code review prompt."""
    code = arguments.get("code", "")
    language = arguments.get("language", "")
    focus = arguments.get("focus", "all")

    focus_instructions = {
        "security": """Focus your review on security concerns:
- Input validation and sanitization
- Authentication and authorization issues
- Potential injection vulnerabilities
- Secure data handling practices""",

        "performance": """Focus your review on performance:
- Algorithm efficiency and complexity
- Memory usage and potential leaks
- Database query optimization
- Caching opportunities""",

        "style": """Focus your review on code style:
- Naming conventions and clarity
- Code organization and structure
- Comments and documentation
- Consistency with language idioms""",

        "all": """Provide a comprehensive review covering:
- Correctness and potential bugs
- Security vulnerabilities
- Performance considerations
- Code style and readability
- Best practices for the language"""
    }

    instruction = focus_instructions.get(focus, focus_instructions["all"])

    text = f"""Please review the following {language} code:

```{language}
{code}
```

{instruction}

For each issue found, please:
1. Identify the location (line or section)
2. Explain the problem
3. Suggest a specific fix or improvement"""

    return {
        "description": f"Code review for {language} ({focus})",
        "messages": [
            {
                "role": "user",
                "content": {"type": "text", "text": text}
            }
        ]
    }


def generate_summarize(arguments: dict) -> dict:
    """Generate a text summarization prompt."""
    text = arguments.get("text", "")
    style = arguments.get("style", "paragraph")
    max_length = arguments.get("max_length", "")

    style_instructions = {
        "paragraph": "Write a cohesive paragraph summary.",
        "bullets": "Summarize as a bulleted list of key points.",
        "tldr": "Provide a single sentence TL;DR summary.",
        "executive": "Write an executive summary with context, key findings, and conclusions."
    }

    instruction = style_instructions.get(style, style_instructions["paragraph"])

    length_note = ""
    if max_length:
        length_note = f" Keep it under {max_length} sentences."

    prompt_text = f"""Please summarize the following text:

---
{text}
---

{instruction}{length_note}

Capture the most important information while maintaining accuracy."""

    return {
        "description": f"Text summary ({style})",
        "messages": [
            {
                "role": "user",
                "content": {"type": "text", "text": prompt_text}
            }
        ]
    }


def generate_explain_code(arguments: dict) -> dict:
    """Generate a code explanation prompt."""
    code = arguments.get("code", "")
    audience = arguments.get("audience", "developer")
    include_examples = arguments.get("include_examples", "false")

    audience_styles = {
        "beginner": "Explain this to someone new to programming. Use simple terms, avoid jargon, and explain any technical concepts.",
        "developer": "Explain this to an experienced developer who may not be familiar with this specific codebase or library.",
        "expert": "Provide a technical deep-dive assuming expertise in the language and related technologies."
    }

    style = audience_styles.get(audience, audience_styles["developer"])

    examples_note = ""
    if include_examples.lower() == "true":
        examples_note = "\n\nInclude example inputs and outputs to illustrate the behavior."

    prompt_text = f"""Please explain what the following code does:

```
{code}
```

{style}{examples_note}

Structure your explanation to cover:
1. Overall purpose - what problem does this solve?
2. How it works - walk through the logic
3. Key components and their roles"""

    return {
        "description": f"Code explanation for {audience}s",
        "messages": [
            {
                "role": "user",
                "content": {"type": "text", "text": prompt_text}
            }
        ]
    }


def generate_git_commit(arguments: dict) -> dict:
    """Generate a git commit message prompt."""
    diff = arguments.get("diff", "")
    style = arguments.get("style", "conventional")

    style_guides = {
        "conventional": """Use conventional commit format:
type(scope): subject

Types: feat, fix, docs, style, refactor, test, chore
Keep subject under 50 characters
Use imperative mood ("add" not "added")""",

        "detailed": """Write a detailed commit message:
- First line: brief summary (under 50 chars)
- Blank line
- Body: explain what and why (not how)
- Wrap at 72 characters""",

        "simple": "Write a clear, single-line commit message under 50 characters."
    }

    guide = style_guides.get(style, style_guides["conventional"])

    prompt_text = f"""Based on the following git diff, write a commit message:

```diff
{diff}
```

{guide}

Analyze the changes and write an appropriate commit message."""

    return {
        "description": f"Git commit message ({style})",
        "messages": [
            {
                "role": "user",
                "content": {"type": "text", "text": prompt_text}
            }
        ]
    }


# Map prompt names to generator functions
GENERATORS = {
    "code-review": generate_code_review,
    "summarize": generate_summarize,
    "explain-code": generate_explain_code,
    "git-commit": generate_git_commit,
}

Section 3: MCP Method Handlers

Add the handlers for MCP methods.

Edit: prompts_server.py

Add the following after the GENERATORS dictionary:

python 
# =============================================================================
# MCP Method Handlers
# =============================================================================

def handle_initialize(params: dict) -> dict:
    """Handle the initialize request."""
    return {
        "protocolVersion": "2024-11-05",
        "capabilities": {"prompts": {}},
        "serverInfo": {"name": "prompts-server", "version": "1.0.0"}
    }


def handle_prompts_list(params: dict) -> dict:
    """List all available prompts."""
    return {"prompts": list(PROMPTS.values())}


def handle_prompts_get(params: dict) -> dict:
    """Get and expand a specific prompt."""
    name = params.get("name")
    arguments = params.get("arguments", {})

    if name not in PROMPTS:
        raise ValueError(f"Unknown prompt: {name}")

    # Validate required arguments
    for arg in PROMPTS[name].get("arguments", []):
        if arg.get("required") and arg["name"] not in arguments:
            raise ValueError(f"Missing required argument: {arg['name']}")

    # Get the generator function
    generator = GENERATORS.get(name)
    if not generator:
        raise ValueError(f"No generator for prompt: {name}")

    return generator(arguments)

Section 4: HTTP Handler

Add the HTTP request handler class.

Edit: prompts_server.py

Add the following after the method handlers:

python 
# =============================================================================
# HTTP Handler
# =============================================================================

class MCPHandler(BaseHTTPRequestHandler):
    """HTTP handler for MCP requests."""

    def log_message(self, format: str, *args) -> None:
        pass  # Suppress default logging

    def do_POST(self) -> None:
        """Handle POST requests."""
        length = int(self.headers.get("Content-Length", 0))
        body = json.loads(self.rfile.read(length).decode())

        method = body.get("method")
        params = body.get("params", {})
        request_id = cast(int | str | None, body.get("id"))

        try:
            if method == "initialize":
                result = handle_initialize(params)
            elif method == "notifications/initialized":
                # Notification - no response needed
                self.send_response(202)
                self.end_headers()
                return
            elif method == "prompts/list":
                result = handle_prompts_list(params)
            elif method == "prompts/get":
                result = handle_prompts_get(params)
            else:
                self.send_error_response(request_id, -32601, f"Unknown method: {method}")
                return

            self.send_json({"jsonrpc": "2.0", "id": request_id, "result": result})

        except ValueError as e:
            self.send_error_response(request_id, -32602, str(e))
        except Exception as e:
            self.send_error_response(request_id, -32603, f"Internal error: {e}")

    def send_json(self, data: dict) -> None:
        """Send a JSON response."""
        body = json.dumps(data, indent=2).encode()
        self.send_response(200)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def send_error_response(self, request_id: Any, code: int, message: str) -> None:
        """Send an error response."""
        self.send_json({
            "jsonrpc": "2.0",
            "id": request_id,
            "error": {"code": code, "message": message}
        })

Section 5: Main Entry Point

Add the main function to start the server.

Edit: prompts_server.py

Add the following at the end of the file:

python 
# =============================================================================
# Main Entry Point
# =============================================================================

def main() -> None:
    """Start the prompts server."""
    host = "localhost"
    port = 8080

    server = HTTPServer((host, port), MCPHandler)
    print(f"Prompts server running on http://{host}:{port}")
    print("Available prompts: code-review, summarize, explain-code, git-commit")
    print("Press Ctrl+C to stop")

    try:
        server.serve_forever()
    except KeyboardInterrupt:
        print("\nShutting down...")
        server.shutdown()


if __name__ == "__main__":
    main()

Prompt Design Tips

Be specific: Give clear instructions on format and scope.
Provide structure: Use numbered steps or sections in prompts.
Handle defaults: Always have sensible defaults for optional arguments.
Test thoroughly: Try your prompts with various inputs.
