Notifications

Learn about fire-and-forget messages that don't expect responses.

What is a Notification?

A notification is a JSON-RPC message that doesn't expect a response. The key difference from a request is the absence of an ID field.

json 
// Request (expects response)
{
    "jsonrpc": "2.0",
    "id": 1,              // Has ID - expects response
    "method": "tools/list",
    "params": {}
}

// Notification (no response expected)
{
    "jsonrpc": "2.0",
                          // No ID field!
    "method": "notifications/initialized",
    "params": {}
}
Fire and Forget

Notifications are "fire and forget" — the sender doesn't wait for or expect any response. The server should NOT send a response to a notification, even if processing fails.

When to Use Notifications

Notifications are ideal for these scenarios:

Use Case Example
Status updates "I'm done initializing"
Progress reports "50% complete"
Log messages "Warning: disk space low"
Resource changes "File X was modified"
Cancellation "Cancel request #5"

MCP Notifications

MCP uses several notifications for lifecycle and event management:

python 
"""Common MCP notifications and their purposes."""

# Client -> Server: "I've finished initializing"
INITIALIZED_NOTIFICATION = {
    "jsonrpc": "2.0",
    "method": "notifications/initialized"
}

# Server -> Client: "My tools have changed, re-fetch them"
TOOLS_CHANGED_NOTIFICATION = {
    "jsonrpc": "2.0",
    "method": "notifications/tools/list_changed"
}

# Server -> Client: "My resources have changed"
RESOURCES_CHANGED_NOTIFICATION = {
    "jsonrpc": "2.0",
    "method": "notifications/resources/list_changed"
}

# Client -> Server: "Cancel this request"
CANCEL_NOTIFICATION = {
    "jsonrpc": "2.0",
    "method": "notifications/cancelled",
    "params": {
        "requestId": 5,
        "reason": "User cancelled operation"
    }
}

# Bidirectional: "Here's a progress update"
PROGRESS_NOTIFICATION = {
    "jsonrpc": "2.0",
    "method": "notifications/progress",
    "params": {
        "progressToken": "task-123",
        "progress": 50,
        "total": 100
    }
}

Creating Notifications

Here's a helper function for creating notifications:

python 
from typing import Optional


def create_notification(method: str, params: dict = None) -> dict:
    """
    Create a JSON-RPC notification (no ID, no response expected).

    Args:
        method: The notification method name
        params: Optional parameters

    Returns:
        A properly formatted JSON-RPC notification
    """
    notification = {
        "jsonrpc": "2.0",
        "method": method
    }

    if params is not None:
        notification["params"] = params

    return notification


# Example usage
init_notification = create_notification("notifications/initialized")
# {"jsonrpc": "2.0", "method": "notifications/initialized"}

progress_notification = create_notification(
    "notifications/progress",
    {"progressToken": "download-1", "progress": 75, "total": 100}
)
# {"jsonrpc": "2.0", "method": "notifications/progress", "params": {...}}

Detecting Notifications

When processing incoming messages, you need to distinguish between requests (need response) and notifications (no response):

python 
def is_notification(message: dict) -> bool:
    """
    Check if a message is a notification.

    A notification is a valid JSON-RPC message without an ID field.
    """
    # Must have jsonrpc and method
    if message.get("jsonrpc") != "2.0":
        return False

    if "method" not in message:
        return False

    # Key distinction: no ID field
    return "id" not in message


def is_request(message: dict) -> bool:
    """
    Check if a message is a request.

    A request is a valid JSON-RPC message WITH an ID field.
    """
    if message.get("jsonrpc") != "2.0":
        return False

    if "method" not in message:
        return False

    # Has ID field
    return "id" in message


def is_response(message: dict) -> bool:
    """
    Check if a message is a response.

    A response has an ID and either result or error (no method).
    """
    if message.get("jsonrpc") != "2.0":
        return False

    if "id" not in message:
        return False

    # Has result or error, but not method
    has_result_or_error = "result" in message or "error" in message
    has_method = "method" in message

    return has_result_or_error and not has_method

Handling Notifications

Process notifications silently — never send a response:

python 
from typing import Optional


class MessageHandler:
    """Handle incoming JSON-RPC messages."""

    def __init__(self):
        self.notification_handlers = {
            "notifications/initialized": self._on_initialized,
            "notifications/cancelled": self._on_cancelled,
            "notifications/progress": self._on_progress,
        }

        self.request_handlers = {
            "initialize": self._handle_initialize,
            "tools/list": self._handle_tools_list,
        }

    def process_message(self, message: dict) -> Optional[dict]:
        """
        Process an incoming message and return response if needed.

        Returns:
            Response dict for requests, None for notifications
        """
        if is_notification(message):
            # Handle notification - NEVER return a response
            self._handle_notification(message)
            return None

        elif is_request(message):
            # Handle request - ALWAYS return a response
            return self._handle_request(message)

        elif is_response(message):
            # Handle response from server (if we're a client)
            self._handle_response(message)
            return None

        else:
            # Invalid message structure
            return create_error_response(
                request_id=message.get("id"),
                code=JsonRpcErrorCode.INVALID_REQUEST,
                message="Invalid message structure"
            )

    def _handle_notification(self, notification: dict):
        """Process a notification silently."""
        method = notification.get("method", "")
        params = notification.get("params", {})

        handler = self.notification_handlers.get(method)
        if handler:
            try:
                handler(params)
            except Exception as e:
                # Log but don't respond
                print(f"Error handling notification {method}: {e}")
        else:
            # Unknown notification - log but don't respond
            print(f"Unknown notification: {method}")

    def _handle_request(self, request: dict) -> dict:
        """Process a request and return a response."""
        request_id = request.get("id")
        method = request.get("method", "")
        params = request.get("params", {})

        handler = self.request_handlers.get(method)
        if not handler:
            return create_error_response(
                request_id=request_id,
                code=JsonRpcErrorCode.METHOD_NOT_FOUND,
                data={"method": method}
            )

        try:
            result = handler(params)
            return {
                "jsonrpc": "2.0",
                "id": request_id,
                "result": result
            }
        except JsonRpcError as e:
            return e.to_response(request_id)

    # Notification handlers (no return value)
    def _on_initialized(self, params: dict):
        print("Client has finished initializing")

    def _on_cancelled(self, params: dict):
        request_id = params.get("requestId")
        print(f"Client cancelled request {request_id}")

    def _on_progress(self, params: dict):
        token = params.get("progressToken")
        progress = params.get("progress", 0)
        total = params.get("total", 100)
        print(f"Progress {token}: {progress}/{total}")

    # Request handlers (return result)
    def _handle_initialize(self, params: dict) -> dict:
        return {"protocolVersion": "2024-11-05", "capabilities": {}}

    def _handle_tools_list(self, params: dict) -> dict:
        return {"tools": []}
Important: No Response!

Never send a response to a notification, even if processing fails. If you send a response, clients may be confused by receiving a response they didn't expect. Log errors server-side instead.

Sending Notifications

From either client or server, notifications are sent without waiting:

python 
import json


class NotificationSender:
    """Helper class for sending notifications."""

    def __init__(self, send_func):
        """
        Args:
            send_func: Function that sends raw JSON string
        """
        self._send = send_func

    def notify(self, method: str, params: dict = None):
        """Send a notification (fire and forget)."""
        notification = create_notification(method, params)
        self._send(json.dumps(notification))
        # Don't wait for response - there won't be one

    def notify_initialized(self):
        """Notify that initialization is complete."""
        self.notify("notifications/initialized")

    def notify_tools_changed(self):
        """Notify that the tools list has changed."""
        self.notify("notifications/tools/list_changed")

    def notify_progress(self, token: str, progress: int, total: int):
        """Send a progress notification."""
        self.notify("notifications/progress", {
            "progressToken": token,
            "progress": progress,
            "total": total
        })

    def notify_cancelled(self, request_id: int, reason: str = None):
        """Notify that a request was cancelled."""
        params = {"requestId": request_id}
        if reason:
            params["reason"] = reason
        self.notify("notifications/cancelled", params)


# Usage example
def send_to_client(data: str):
    print(f"Sending: {data}")

sender = NotificationSender(send_to_client)
sender.notify_progress("upload-1", 50, 100)
# Sending: {"jsonrpc": "2.0", "method": "notifications/progress", "params": {...}}

Request vs Notification Decision Tree

Use this guide to decide whether to use a request or notification:

text 
Do you need a response?
    |
    +-- YES --> Use a Request (include ID)
    |           Examples: tools/list, tools/call, initialize
    |
    +-- NO --> Use a Notification (no ID)
               |
               +-- Is it a status update? --> notifications/initialized
               +-- Is it a change event? --> notifications/tools/list_changed
               +-- Is it progress info? --> notifications/progress
               +-- Is it cancellation? --> notifications/cancelled
