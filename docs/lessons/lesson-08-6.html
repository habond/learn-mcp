Complete HTTP Server

Put it all together with a full working MCP server using HTTP/SSE transport.

Putting It All Together

Now let's combine everything we've learned into a complete, production-ready MCP server using HTTP/SSE transport. This server will:

  • Handle the MCP initialization handshake over HTTP
  • Support both JSON and SSE response modes
  • Manage client sessions with the Mcp-Session-Id header
  • Expose tools that can stream progress via SSE
  • Include comprehensive error handling and logging

The Complete Server

Save this as sse_server.py:

python 
#!/usr/bin/env python3
"""
Complete MCP Server using HTTP/SSE Transport.

This server demonstrates the Streamable HTTP transport for MCP:
- POST /mcp: JSON-RPC endpoint with optional SSE streaming
- Session management via Mcp-Session-Id header
- Support for both immediate JSON and streaming SSE responses

Usage:
    python sse_server.py

Then test with:
    curl -X POST http://localhost:8080/mcp \
         -H "Content-Type: application/json" \
         -d '{"jsonrpc":"2.0","id":1,"method":"initialize","params":{...}}'
"""

import json
import uuid
import time
import logging
from http.server import HTTPServer, BaseHTTPRequestHandler
from datetime import datetime
from dataclasses import dataclass, field
from typing import Dict, Optional, Any, Generator

# ============================================================================
# Logging Setup
# ============================================================================

logging.basicConfig(
    level=logging.DEBUG,
    format='%(asctime)s [%(levelname)s] %(message)s'
)
logger = logging.getLogger("mcp-sse-server")

# ============================================================================
# JSON-RPC Error Codes
# ============================================================================

class ErrorCode:
    PARSE_ERROR = -32700
    INVALID_REQUEST = -32600
    METHOD_NOT_FOUND = -32601
    INVALID_PARAMS = -32602
    INTERNAL_ERROR = -32603

# ============================================================================
# Session Management
# ============================================================================

@dataclass
class MCPSession:
    """Represents an MCP client session."""
    session_id: str
    created_at: float = field(default_factory=time.time)
    last_active: float = field(default_factory=time.time)
    initialized: bool = False
    client_info: Dict[str, Any] = field(default_factory=dict)
    capabilities: Dict[str, Any] = field(default_factory=dict)

    def touch(self):
        """Update last active time."""
        self.last_active = time.time()

    def is_expired(self, timeout_seconds: int = 3600) -> bool:
        """Check if session has expired."""
        return time.time() - self.last_active > timeout_seconds


class SessionStore:
    """Store and manage MCP sessions."""

    def __init__(self, session_timeout: int = 3600):
        self.sessions: Dict[str, MCPSession] = {}
        self.session_timeout = session_timeout

    def create_session(self) -> MCPSession:
        """Create a new session."""
        session_id = str(uuid.uuid4())
        session = MCPSession(session_id=session_id)
        self.sessions[session_id] = session
        logger.info(f"Created session: {session_id}")
        return session

    def get_session(self, session_id: str) -> Optional[MCPSession]:
        """Get a session by ID."""
        session = self.sessions.get(session_id)
        if session is None:
            return None
        if session.is_expired(self.session_timeout):
            self.delete_session(session_id)
            return None
        session.touch()
        return session

    def delete_session(self, session_id: str) -> bool:
        """Delete a session."""
        if session_id in self.sessions:
            del self.sessions[session_id]
            logger.info(f"Deleted session: {session_id}")
            return True
        return False


# Global session store
sessions = SessionStore()

# ============================================================================
# Tool Definitions
# ============================================================================

TOOLS = [
    {
        "name": "echo",
        "description": "Echo back a message. Useful for testing.",
        "inputSchema": {
            "type": "object",
            "properties": {
                "message": {
                    "type": "string",
                    "description": "The message to echo back"
                }
            },
            "required": ["message"]
        }
    },
    {
        "name": "get_time",
        "description": "Get the current server time.",
        "inputSchema": {
            "type": "object",
            "properties": {},
            "required": []
        }
    },
    {
        "name": "countdown",
        "description": "Count down from a number (demonstrates streaming).",
        "inputSchema": {
            "type": "object",
            "properties": {
                "start": {
                    "type": "integer",
                    "description": "Number to count down from",
                    "minimum": 1,
                    "maximum": 10
                }
            },
            "required": ["start"]
        }
    },
    {
        "name": "add_numbers",
        "description": "Add two numbers together.",
        "inputSchema": {
            "type": "object",
            "properties": {
                "a": {"type": "number", "description": "First number"},
                "b": {"type": "number", "description": "Second number"}
            },
            "required": ["a", "b"]
        }
    }
]


def execute_tool(name: str, arguments: dict) -> dict:
    """Execute a tool and return the result."""
    logger.info(f"Executing tool: {name} with args: {arguments}")

    if name == "echo":
        message = arguments.get("message", "")
        return {"content": [{"type": "text", "text": message}]}

    elif name == "get_time":
        current_time = datetime.now().isoformat()
        return {"content": [{"type": "text", "text": f"Current time: {current_time}"}]}

    elif name == "countdown":
        start = arguments.get("start", 5)
        result = " -> ".join(str(i) for i in range(start, 0, -1)) + " -> Done!"
        return {"content": [{"type": "text", "text": result}]}

    elif name == "add_numbers":
        a = arguments.get("a", 0)
        b = arguments.get("b", 0)
        result = a + b
        return {"content": [{"type": "text", "text": f"{a} + {b} = {result}"}]}

    else:
        raise ValueError(f"Unknown tool: {name}")


def execute_tool_streaming(name: str, arguments: dict, request_id) -> Generator[str, None, None]:
    """
    Execute a tool with SSE streaming support.

    Yields SSE-formatted events for progress and final result.
    """
    if name == "countdown":
        start = arguments.get("start", 5)

        # Send progress notifications
        for i in range(start, 0, -1):
            progress_notification = {
                "jsonrpc": "2.0",
                "method": "notifications/progress",
                "params": {
                    "progressToken": request_id,
                    "progress": start - i,
                    "total": start,
                    "message": f"Counting: {i}"
                }
            }
            yield f"event: message\ndata: {json.dumps(progress_notification)}\n\n"
            time.sleep(0.5)  # Simulate work

        # Send final result
        result = {
            "jsonrpc": "2.0",
            "id": request_id,
            "result": {
                "content": [{"type": "text", "text": "Countdown complete!"}]
            }
        }
        yield f"event: message\ndata: {json.dumps(result)}\n\n"
    else:
        # Non-streaming tools just return result immediately
        result = execute_tool(name, arguments)
        response = {
            "jsonrpc": "2.0",
            "id": request_id,
            "result": result
        }
        yield f"event: message\ndata: {json.dumps(response)}\n\n"

# ============================================================================
# HTTP Request Handler
# ============================================================================

class MCPHTTPHandler(BaseHTTPRequestHandler):
    """HTTP handler for MCP with SSE support."""

    protocol_version = "HTTP/1.1"

    def log_message(self, format, *args):
        """Override to use our logger."""
        logger.debug(f"{self.address_string()} - {format % args}")

    def get_session(self) -> Optional[MCPSession]:
        """Get session from request header."""
        session_id = self.headers.get("Mcp-Session-Id")
        if session_id:
            return sessions.get_session(session_id)
        return None

    def wants_sse(self) -> bool:
        """Check if client accepts SSE responses."""
        accept = self.headers.get("Accept", "")
        return "text/event-stream" in accept

    def send_cors_headers(self):
        """Send CORS headers for browser clients."""
        self.send_header("Access-Control-Allow-Origin", "*")
        self.send_header("Access-Control-Allow-Methods", "POST, DELETE, OPTIONS")
        self.send_header("Access-Control-Allow-Headers",
                        "Content-Type, Accept, Mcp-Session-Id")
        self.send_header("Access-Control-Expose-Headers", "Mcp-Session-Id")

    def do_OPTIONS(self):
        """Handle CORS preflight requests."""
        self.send_response(204)
        self.send_cors_headers()
        self.end_headers()

    def do_DELETE(self):
        """Handle session termination."""
        if self.path != "/mcp":
            self.send_error(404, "Not Found")
            return

        session = self.get_session()
        if session:
            sessions.delete_session(session.session_id)
            self.send_response(204)
            self.send_cors_headers()
            self.end_headers()
        else:
            self.send_error(404, "Session not found")

    def do_POST(self):
        """Handle MCP JSON-RPC requests."""
        if self.path != "/mcp":
            self.send_error(404, "Not Found")
            return

        # Read and parse request body
        content_length = int(self.headers.get("Content-Length", 0))
        body = self.rfile.read(content_length).decode("utf-8")

        try:
            request = json.loads(body)
        except json.JSONDecodeError as e:
            self.send_json_response({
                "jsonrpc": "2.0",
                "id": None,
                "error": {"code": ErrorCode.PARSE_ERROR, "message": str(e)}
            }, status=400)
            return

        # Validate JSON-RPC format
        if request.get("jsonrpc") != "2.0":
            self.send_json_response({
                "jsonrpc": "2.0",
                "id": request.get("id"),
                "error": {"code": ErrorCode.INVALID_REQUEST,
                         "message": "Invalid JSON-RPC version"}
            }, status=400)
            return

        method = request.get("method")
        request_id = request.get("id")
        params = request.get("params", {})

        logger.debug(f"Request: {method} (id={request_id})")

        # Handle initialize (creates new session)
        if method == "initialize":
            self.handle_initialize(request_id, params)
            return

        # All other methods require a valid session
        session = self.get_session()
        if session is None:
            self.send_json_response({
                "jsonrpc": "2.0",
                "id": request_id,
                "error": {"code": ErrorCode.INVALID_REQUEST,
                         "message": "No valid session. Call 'initialize' first."}
            }, status=400)
            return

        # Handle notifications (no response needed)
        if request_id is None:
            self.handle_notification(method, params, session)
            self.send_response(202)
            self.send_cors_headers()
            self.send_header("Mcp-Session-Id", session.session_id)
            self.end_headers()
            return

        # Check if session is initialized for most methods
        if not session.initialized and method != "notifications/initialized":
            self.send_json_response({
                "jsonrpc": "2.0",
                "id": request_id,
                "error": {"code": ErrorCode.INVALID_REQUEST,
                         "message": "Session not initialized"}
            }, status=400, session=session)
            return

        # Route to appropriate handler
        if method == "tools/list":
            self.handle_tools_list(request_id, session)
        elif method == "tools/call":
            self.handle_tools_call(request_id, params, session)
        elif method == "resources/list":
            self.handle_resources_list(request_id, session)
        elif method == "prompts/list":
            self.handle_prompts_list(request_id, session)
        elif method == "ping":
            self.handle_ping(request_id, session)
        else:
            self.send_json_response({
                "jsonrpc": "2.0",
                "id": request_id,
                "error": {"code": ErrorCode.METHOD_NOT_FOUND,
                         "message": f"Method not found: {method}"}
            }, session=session)

    # ========================================================================
    # MCP Method Handlers
    # ========================================================================

    def handle_initialize(self, request_id, params):
        """Handle the initialize request."""
        # Create new session
        session = sessions.create_session()
        session.client_info = params.get("clientInfo", {})
        session.capabilities = params.get("capabilities", {})

        client_name = session.client_info.get("name", "unknown")
        logger.info(f"Client connecting: {client_name}")

        response = {
            "jsonrpc": "2.0",
            "id": request_id,
            "result": {
                "protocolVersion": "2024-11-05",
                "capabilities": {
                    "tools": {},
                    "resources": {},
                    "prompts": {}
                },
                "serverInfo": {
                    "name": "sse-demo-server",
                    "version": "1.0.0"
                }
            }
        }

        self.send_json_response(response, session=session)

    def handle_notification(self, method, params, session):
        """Handle notifications (no response expected)."""
        if method == "notifications/initialized":
            session.initialized = True
            logger.info(f"Session initialized: {session.session_id}")
        elif method == "notifications/cancelled":
            logger.info(f"Request cancelled: {params}")
        else:
            logger.warning(f"Unknown notification: {method}")

    def handle_tools_list(self, request_id, session):
        """Handle the tools/list request."""
        self.send_json_response({
            "jsonrpc": "2.0",
            "id": request_id,
            "result": {"tools": TOOLS}
        }, session=session)

    def handle_tools_call(self, request_id, params, session):
        """Handle the tools/call request with optional SSE streaming."""
        tool_name = params.get("name")
        arguments = params.get("arguments", {})

        # Check if tool exists
        if not any(t["name"] == tool_name for t in TOOLS):
            self.send_json_response({
                "jsonrpc": "2.0",
                "id": request_id,
                "error": {"code": ErrorCode.METHOD_NOT_FOUND,
                         "message": f"Unknown tool: {tool_name}"}
            }, session=session)
            return

        # Check if client wants SSE streaming
        if self.wants_sse() and tool_name == "countdown":
            self.send_sse_response(
                execute_tool_streaming(tool_name, arguments, request_id),
                session
            )
        else:
            # Standard JSON response
            try:
                result = execute_tool(tool_name, arguments)
                self.send_json_response({
                    "jsonrpc": "2.0",
                    "id": request_id,
                    "result": result
                }, session=session)
            except Exception as e:
                logger.error(f"Tool execution error: {e}")
                self.send_json_response({
                    "jsonrpc": "2.0",
                    "id": request_id,
                    "error": {"code": ErrorCode.INTERNAL_ERROR,
                             "message": str(e)}
                }, session=session)

    def handle_resources_list(self, request_id, session):
        """Handle the resources/list request."""
        self.send_json_response({
            "jsonrpc": "2.0",
            "id": request_id,
            "result": {"resources": []}
        }, session=session)

    def handle_prompts_list(self, request_id, session):
        """Handle the prompts/list request."""
        self.send_json_response({
            "jsonrpc": "2.0",
            "id": request_id,
            "result": {"prompts": []}
        }, session=session)

    def handle_ping(self, request_id, session):
        """Handle ping request."""
        self.send_json_response({
            "jsonrpc": "2.0",
            "id": request_id,
            "result": {}
        }, session=session)

    # ========================================================================
    # Response Helpers
    # ========================================================================

    def send_json_response(self, data, status=200, session=None):
        """Send a JSON response."""
        body = json.dumps(data).encode("utf-8")
        self.send_response(status)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        self.send_cors_headers()
        if session:
            self.send_header("Mcp-Session-Id", session.session_id)
        self.end_headers()
        self.wfile.write(body)

    def send_sse_response(self, event_generator: Generator[str, None, None], session):
        """Send an SSE streaming response."""
        self.send_response(200)
        self.send_header("Content-Type", "text/event-stream")
        self.send_header("Cache-Control", "no-cache")
        self.send_header("Connection", "keep-alive")
        self.send_cors_headers()
        self.send_header("Mcp-Session-Id", session.session_id)
        self.end_headers()

        try:
            for event in event_generator:
                self.wfile.write(event.encode("utf-8"))
                self.wfile.flush()
        except (BrokenPipeError, ConnectionResetError):
            logger.warning("Client disconnected during SSE stream")

# ============================================================================
# Main Entry Point
# ============================================================================

def main():
    """Start the MCP HTTP/SSE server."""
    host = "localhost"
    port = 8080

    server = HTTPServer((host, port), MCPHTTPHandler)
    logger.info(f"MCP HTTP/SSE Server starting on http://{host}:{port}")
    logger.info("Endpoints:")
    logger.info(f"  POST   http://{host}:{port}/mcp  - JSON-RPC requests")
    logger.info(f"  DELETE http://{host}:{port}/mcp  - Terminate session")
    logger.info("")
    logger.info("Press Ctrl+C to stop")

    try:
        server.serve_forever()
    except KeyboardInterrupt:
        logger.info("Shutting down...")
        server.shutdown()


if __name__ == "__main__":
    main()

Production Considerations
  • Use HTTPS in production (TLS termination via reverse proxy)
  • Implement rate limiting to prevent abuse
  • Add authentication (API keys, OAuth, etc.)
  • Use a production WSGI server like Gunicorn or uWSGI
  • Store sessions in Redis for horizontal scaling
  • Implement proper session expiration and cleanup

HTTP/SSE vs STDIO Comparison

Feature STDIO HTTP/SSE
Connection Local process Network (local or remote)
Multiple Clients One per process Many concurrent
Session Management Implicit (process lifetime) Explicit (Mcp-Session-Id)
Streaming Line-by-line SSE events
Complexity Simpler More complex
Use Case Desktop apps, CLI tools Web services, shared servers
