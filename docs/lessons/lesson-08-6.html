# =============================================================================
# Section 2: HTTP Request Handler
# =============================================================================

class MCPHTTPHandler(BaseHTTPRequestHandler):
    """HTTP handler for MCP requests."""

    protocol_version = "HTTP/1.1"

    def log_message(self, format: str, *args) -> None:
        """Use our logger instead of default."""
        logger.debug(f"{self.address_string()} - {format % args}")

    def send_cors_headers(self) -> None:
        """Send CORS headers for browser clients."""
        self.send_header("Access-Control-Allow-Origin", "*")
        self.send_header("Access-Control-Allow-Methods", "POST, OPTIONS")
        self.send_header("Access-Control-Allow-Headers", "Content-Type, Accept")

    def do_OPTIONS(self) -> None:
        """Handle CORS preflight requests."""
        self.send_response(204)
        self.send_cors_headers()
        self.end_headers()

    def send_json(self, data: dict, status: int = 200) -> None:
        """Send a JSON response."""
        body = json.dumps(data, indent=2).encode("utf-8")
        self.send_response(status)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        self.send_cors_headers()
        self.end_headers()
        self.wfile.write(body)

    def do_POST(self) -> None:
        """Handle MCP JSON-RPC requests."""
        if self.path != "/mcp":
            self.send_error(404, "Not Found")
            return

        # Read request body
        content_length = int(self.headers.get("Content-Length", 0))
        body = self.rfile.read(content_length).decode("utf-8")

        try:
            request = json.loads(body)
        except json.JSONDecodeError as e:
            self.send_json(create_error_response(None, JsonRpcErrorCode.PARSE_ERROR, str(e)), 400)
            return

        # Use shared handler from mcp_core
        response = handle_request(request)

        if response is None:
            # Notification - no response body
            self.send_response(204)
            self.send_cors_headers()
            self.end_headers()
        else:
            self.send_json(response)

# =============================================================================
# Section 3: Server Runner

def run_http_server(host: str = "localhost", port: int = 8080) -> None:
    """
    Run MCP server using HTTP transport.

    Args:
        host: Host to bind to (default: localhost)
        port: Port to listen on (default: 8080)
    """
    server = HTTPServer((host, port), MCPHTTPHandler)
    logger.info(f"MCP HTTP Server running on http://{host}:{port}")
    logger.info(f"Endpoint: POST http://{host}:{port}/mcp")
    logger.info("Press Ctrl+C to stop")

    try:
        server.serve_forever()
    except KeyboardInterrupt:
        logger.info("Shutting down...")
        server.shutdown()


if __name__ == "__main__":
    run_http_server()

Section 4: Example with Tools

Create: http_example.py

Create a server that uses HTTP transport with tools:

python http_example.py 
"""
http_example.py - HTTP server using tools.py

Run: python http_example.py
Test: curl -X POST http://localhost:8080/mcp ...
"""

from datetime import datetime

# Import from our modules
from mcp_core import router
from tools import ToolRegistry, register_tool_handlers, text_result
from transports.http_sse import run_http_server

# =============================================================================
# Register tools (same as stdio_example.py!)
# =============================================================================

tool_registry = ToolRegistry()


@tool_registry.register(
    name="echo",
    description="Echo back a message",
    input_schema={
        "type": "object",
        "properties": {"message": {"type": "string", "description": "Message to echo"}},
        "required": ["message"]
    }
)
def echo(args: dict) -> dict:
    return text_result(args.get("message", ""))


@tool_registry.register(
    name="get_time",
    description="Get current server time",
    input_schema={"type": "object", "properties": {}, "required": []}
)
def get_time(args: dict) -> dict:
    return text_result(f"Current time: {datetime.now().isoformat()}")


@tool_registry.register(
    name="add_numbers",
    description="Add two numbers",
    input_schema={
        "type": "object",
        "properties": {
            "a": {"type": "number", "description": "First number"},
            "b": {"type": "number", "description": "Second number"}
        },
        "required": ["a", "b"]
    }
)
def add_numbers(args: dict) -> dict:
    a = args.get("a", 0)
    b = args.get("b", 0)
    return text_result(f"{a} + {b} = {a + b}")


# Register with the MCP router
register_tool_handlers(router, tool_registry)


# =============================================================================
# Run the server
# =============================================================================

if __name__ == "__main__":
    run_http_server()

Transport Independence

Notice how http_example.py and stdio_example.py register the same tools. The only difference is which transport's run_* function they call. This is the power of the modular architecture!

Production Considerations
  • Use HTTPS in production (via reverse proxy)
  • Implement rate limiting and authentication
  • Use a production server like Gunicorn or uWSGI
