Building File Resources

Build a complete MCP server that exposes local files as resources.

Building Incrementally

Let's build a complete file resource server step by step. We'll create file_resource_server.py incrementally, testing as we go.

File Structure

We'll build one file: file_resource_server.py. Each section below adds to this file. Follow along and add each section in order.

Section 1: Imports and Configuration

Create: file_resource_server.py

Start with the imports and configuration:

python file_resource_server.py 
"""
file_resource_server.py - MCP server exposing files as resources.

Run with: python file_resource_server.py
Test with: curl commands
"""

import json
import base64
import mimetypes
from pathlib import Path
from http.server import HTTPServer, BaseHTTPRequestHandler
from typing import cast


# =============================================================================
# Section 1: Configuration
# =============================================================================

PROJECT_DIR = Path("./project")  # Directory to expose
ALLOWED_EXTENSIONS = {".py", ".js", ".json", ".md", ".txt", ".html", ".css"}


class MCPError(Exception):
    """Custom exception for MCP errors."""
    def __init__(self, code: int, message: str):
        self.code = code
        self.message = message
        super().__init__(message)


# Quick test

def get_mime_type(file_path: Path) -> str:
    """Get MIME type for a file."""
    mime_type, _ = mimetypes.guess_type(str(file_path))
    if mime_type:
        return mime_type

    # Custom mappings for common code files
    custom_types = {
        ".py": "text/x-python",
        ".js": "text/javascript",
        ".ts": "text/typescript",
        ".md": "text/markdown",
        ".yml": "text/yaml",
        ".yaml": "text/yaml",
    }
    return custom_types.get(file_path.suffix.lower(), "text/plain")


def is_binary_file(file_path: Path) -> bool:
    """Check if a file is binary."""
    binary_extensions = {".png", ".jpg", ".jpeg", ".gif", ".pdf", ".zip", ".exe"}
    return file_path.suffix.lower() in binary_extensions

def scan_directory(directory: Path) -> list[dict]:
    """Scan directory and return resource definitions."""
    resources = []

    if not directory.exists():
        return resources

    for file_path in directory.rglob("*"):
        if not file_path.is_file():
            continue

        # Skip files with disallowed extensions
        if ALLOWED_EXTENSIONS and file_path.suffix.lower() not in ALLOWED_EXTENSIONS:
            continue

        # Skip hidden files and directories
        if any(part.startswith(".") for part in file_path.parts):
            continue

        # Create resource definition
        uri = f"file://{file_path.absolute()}"
        relative_path = file_path.relative_to(directory)

        resources.append({
            "uri": uri,
            "name": file_path.name,
            "description": f"File: {relative_path}",
            "mimeType": get_mime_type(file_path)
        })

    return sorted(resources, key=lambda r: r["name"])

# =============================================================================
# Section 4: File Reader
# =============================================================================

def read_file_resource(uri: str) -> dict:
    """Read a file and return its content."""
    if not uri.startswith("file://"):
        raise MCPError(-32602, f"Invalid URI scheme: {uri}")

    file_path = Path(uri[7:])  # Remove "file://" prefix

    if not file_path.exists():
        raise MCPError(-32002, f"File not found: {uri}")

    if not file_path.is_file():
        raise MCPError(-32002, f"Not a file: {uri}")

    # Security check: ensure file is within allowed directory
    try:
        file_path.resolve().relative_to(PROJECT_DIR.resolve())
    except ValueError:
        raise MCPError(-32003, f"Access denied: {uri}")

    mime_type = get_mime_type(file_path)

    if is_binary_file(file_path):
        # Read as binary and encode as base64
        with open(file_path, "rb") as f:
            content = base64.b64encode(f.read()).decode("utf-8")
        return {
            "uri": uri,
            "mimeType": mime_type,
            "blob": content
        }
    else:
        # Read as text
        try:
            with open(file_path, "r", encoding="utf-8") as f:
                content = f.read()
        except UnicodeDecodeError:
            with open(file_path, "rb") as f:
                content = base64.b64encode(f.read()).decode("utf-8")
            return {
                "uri": uri,
                "mimeType": "application/octet-stream",
                "blob": content
            }

        return {
            "uri": uri,
            "mimeType": mime_type,
            "text": content
        }

def handle_initialize(params: dict) -> dict:
    """Handle initialize request."""
    return {
        "protocolVersion": "2024-11-05",
        "capabilities": {
            "resources": {}  # We support resources
        },
        "serverInfo": {
            "name": "file-resource-server",
            "version": "1.0.0"
        }
    }


def handle_resources_list(params: dict) -> dict:
    """Handle resources/list request."""
    resources = scan_directory(PROJECT_DIR)
    return {"resources": resources}


def handle_resources_read(params: dict) -> dict:
    """Handle resources/read request."""
    uri = params.get("uri")
    if not uri:
        raise MCPError(-32602, "Missing required parameter: uri")

    content = read_file_resource(uri)
    return {"contents": [content]}

# =============================================================================
# Section 6: Request Router and HTTP Server
# =============================================================================

def route_request(method: str, params: dict) -> dict | None:
    """Route JSON-RPC requests to handlers."""
    handlers = {
        "initialize": handle_initialize,
        "resources/list": handle_resources_list,
        "resources/read": handle_resources_read,
    }

    handler = handlers.get(method)
    if handler:
        return handler(params)

    # Handle notifications
    if method == "notifications/initialized":
        return None

    raise MCPError(-32601, f"Method not found: {method}")


class MCPHandler(BaseHTTPRequestHandler):
    """HTTP handler for MCP requests."""

    def log_message(self, format, *args):
        """Custom logging."""
        print(f"[MCP] {args[0]}")

    def send_json(self, data: dict, status: int = 200):
        """Send JSON response."""
        body = json.dumps(data, indent=2).encode("utf-8")
        self.send_response(status)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def do_POST(self):
        """Handle POST requests."""
        try:
            content_length = int(self.headers.get("Content-Length", 0))
            body = self.rfile.read(content_length)
            request = json.loads(body.decode("utf-8"))

            method = request.get("method")
            params = request.get("params", {})
            request_id = cast(int | str | None, request.get("id"))

            # Handle notification (no id)
            if request_id is None:
                route_request(method, params)
                self.send_response(204)
                self.end_headers()
                return

            # Handle request
            try:
                result = route_request(method, params)
                if result is None:
                    self.send_response(204)
                    self.end_headers()
                    return
                self.send_json({
                    "jsonrpc": "2.0",
                    "id": request_id,
                    "result": result
                })
            except MCPError as e:
                self.send_json({
                    "jsonrpc": "2.0",
                    "id": request_id,
                    "error": {"code": e.code, "message": e.message}
                })

        except json.JSONDecodeError as e:
            self.send_json({
                "jsonrpc": "2.0",
                "id": None,
                "error": {"code": -32700, "message": f"Parse error: {e}"}
            }, status=400)


def run_server(host: str = "localhost", port: int = 8080):
    """Start the MCP file resource server."""
    # Create project directory with sample files
    PROJECT_DIR.mkdir(exist_ok=True)
    (PROJECT_DIR / "config.json").write_text('{\n  "debug": true,\n  "port": 8080\n}')
    (PROJECT_DIR / "README.md").write_text("# My Project\n\nThis is a sample project.")
    (PROJECT_DIR / "main.py").write_text('def main():\n    print("Hello!")\n\nif __name__ == "__main__":\n    main()')

    server = HTTPServer((host, port), MCPHandler)
    print(f"File Resource Server running on http://{host}:{port}")
    print(f"Serving files from: {PROJECT_DIR.absolute()}")
    print("Press Ctrl+C to stop")

    try:
        server.serve_forever()
    except KeyboardInterrupt:
        print("\nShutting down...")
        server.shutdown()

What You've Built

Your file resource server now:

  • Scans a directory for files with allowed extensions
  • Exposes them as MCP resources with proper metadata
  • Returns text content for readable files
  • Returns base64-encoded blobs for binary files
  • Includes security checks to prevent directory traversal
  • Handles errors gracefully with proper JSON-RPC error responses
Extension Ideas
  • Add file watching to update resources when files change
  • Implement resource subscriptions for real-time updates
  • Add support for resource templates (dynamic URIs)
  • Include file metadata like size and modification time
