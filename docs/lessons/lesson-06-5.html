Section 1: Create resources.py

Create: resources.py

Create a new file with imports from Lesson 3:

python resources.py 
"""
resources.py - Reusable resource registry for MCP servers.

This module builds on jsonrpc.py from Lesson 3.
"""

from typing import Callable

# Import error handling from Lesson 3
from jsonrpc import JsonRpcError, JsonRpcErrorCode


# =============================================================================
# Section 1: Resource Registry
# =============================================================================

class ResourceRegistry:
    """
    Registry for MCP resources with decorator-based registration.

    Usage:
        resources = ResourceRegistry()

        @resources.register(
            uri="file://config.json",
            name="Config",
            description="Configuration file",
            mime_type="application/json"
        )
        def read_config():
            return {"uri": "file://config.json", "mimeType": "application/json", "text": "{}"}
    """

    def __init__(self):
        self._resources: dict[str, dict] = {}
        self._readers: dict[str, Callable] = {}

    def register(
        self,
        uri: str,
        name: str,
        description: str,
        mime_type: str = "text/plain"
    ) -> Callable:
        """Decorator to register a resource reader."""
        def decorator(func: Callable[[], dict]) -> Callable:
            self._resources[uri] = {
                "uri": uri,
                "name": name,
                "description": description,
                "mimeType": mime_type
            }
            self._readers[uri] = func
            return func
        return decorator

    def list_resources(self) -> list[dict]:
        """Return all resource definitions for resources/list."""
        return list(self._resources.values())

    def read_resource(self, uri: str) -> dict:
        """Read a resource by URI."""
        if uri not in self._readers:
            raise JsonRpcError(
                JsonRpcErrorCode.INVALID_PARAMS,
                f"Resource not found: {uri}"
            )
        content = self._readers[uri]()
        return {"contents": [content]}

Section 2: Router Integration

Edit: resources.py

Add router integration after Section 1:

python resources.py - Add after Section 1 
# =============================================================================
# Section 2: MCP Router Integration
# =============================================================================

def register_resource_handlers(router, resources: ResourceRegistry) -> None:
    """
    Register resources/list and resources/read handlers with an MCP router.

    Args:
        router: A JsonRpcRouter from mcp_core.py
        resources: A ResourceRegistry with registered resources
    """

    @router.method("resources/list")
    def handle_resources_list(params: dict) -> dict:
        return {"resources": resources.list_resources()}

    @router.method("resources/read")
    def handle_resources_read(params: dict) -> dict:
        uri = params.get("uri", "")
        return resources.read_resource(uri)

Section 3: File Example

Create: file_example.py

Create a file server that uses resources.py and mcp_core.py:

python file_example.py 
"""
file_example.py - File server using resources.py and mcp_core.py

Run: python file_example.py
"""

import json
from pathlib import Path
from http.server import HTTPServer, BaseHTTPRequestHandler

# Import from our modules
from jsonrpc import JsonRpcErrorCode, create_error_response
from mcp_core import router, handle_request
from resources import ResourceRegistry, register_resource_handlers

# =============================================================================
# Create resource registry and register file resources
# =============================================================================

file_resources = ResourceRegistry()


@file_resources.register(
    uri="file://readme",
    name="README",
    description="Project readme file",
    mime_type="text/markdown"
)
def read_readme():
    path = Path("README.md")
    if path.exists():
        return {"uri": "file://readme", "mimeType": "text/markdown", "text": path.read_text()}
    return {"uri": "file://readme", "mimeType": "text/plain", "text": "No README found"}


@file_resources.register(
    uri="file://package",
    name="package.json",
    description="Package configuration",
    mime_type="application/json"
)
def read_package():
    path = Path("package.json")
    if path.exists():
        return {"uri": "file://package", "mimeType": "application/json", "text": path.read_text()}
    return {"uri": "file://package", "mimeType": "application/json", "text": "{}"}


# Register resource handlers with the MCP router
register_resource_handlers(router, file_resources)


# =============================================================================
# HTTP Server
# =============================================================================

class MCPHandler(BaseHTTPRequestHandler):
    def log_message(self, format, *args):
        print(f"[Files] {args[0]}")

    def send_json(self, data: dict, status: int = 200):
        body = json.dumps(data, indent=2).encode()
        self.send_response(status)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def do_POST(self):
        length = int(self.headers.get("Content-Length", 0))
        body = self.rfile.read(length)

        try:
            request = json.loads(body)
        except json.JSONDecodeError as e:
            self.send_json(create_error_response(None, JsonRpcErrorCode.PARSE_ERROR, str(e)), 400)
            return

        response = handle_request(request)
        if response is None:
            self.send_response(204)
            self.end_headers()
        else:
            self.send_json(response)


if __name__ == "__main__":
    print("File Resource Server")
    print("Resources: readme, package")
    print("Running on http://localhost:8080\n")

    server = HTTPServer(("localhost", 8080), MCPHandler)
    try:
        server.serve_forever()
    except KeyboardInterrupt:
        print("\nShutting down...")

What You've Built

You've created a reusable resource system:

File Purpose
resources.py Reusable ResourceRegistry with decorator pattern
file_example.py Example server using all previous modules

The pattern mirrors tools.py from Lesson 5:

  • Decorator-based registration for clean resource definitions
  • Imports from jsonrpc.py for error handling
  • Works with mcp_core.py router
  • Can be combined with tools in the same server
Combining Tools and Resources

You can combine tools.py and resources.py in one server. Just import both and call both registration functions with the same router.
