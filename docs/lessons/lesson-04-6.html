Building a Complete Handshake Server

Let's bring together everything we've learned in this lesson: the MCP lifecycle, the initialize request, server response with capabilities, and the initialized notification. This complete server properly implements the full handshake sequence.

python 
"""
handshake_server.py - Complete MCP server with proper handshake handling.

This server demonstrates:
- Connection state management (DISCONNECTED -> INITIALIZING -> READY)
- Protocol version negotiation
- Capability exchange
- Proper notification handling
- Request validation based on connection state

Run with: python handshake_server.py
Test with: curl commands below
"""

import json
from http.server import HTTPServer, BaseHTTPRequestHandler
from enum import Enum
from datetime import datetime


# =============================================================================
# Connection State Management
# =============================================================================

class ConnectionState(Enum):
    """Track the state of an MCP connection."""
    DISCONNECTED = "disconnected"  # Initial state, waiting for initialize
    INITIALIZING = "initializing"  # Received initialize, waiting for initialized
    READY = "ready"                # Handshake complete, normal operation


class MCPConnection:
    """Manages state and metadata for an MCP connection."""

    # Supported protocol version
    PROTOCOL_VERSION = "2024-11-05"

    def __init__(self):
        self.state = ConnectionState.DISCONNECTED
        self.client_info = None
        self.client_capabilities = None
        self.negotiated_version = None
        self.connected_at = None
        self.ready_at = None

    def reset(self):
        """Reset connection to initial state."""
        self.__init__()

    def is_ready(self) -> bool:
        """Check if connection is ready for normal operations."""
        return self.state == ConnectionState.READY


# Global connection instance (in production, you'd manage per-client)
connection = MCPConnection()


# =============================================================================
# Server Capabilities
# =============================================================================

def get_server_capabilities():
    """Return the capabilities this server supports.

    This is where you advertise what features your server offers.
    """
    return {
        "tools": {},           # We support tools (will implement in Lesson 5)
        "resources": {},       # We support resources (will implement in Lesson 6)
        # "prompts": {},       # Uncomment when you add prompt support
        # "logging": {},       # Uncomment when you add logging support
    }


def get_server_info():
    """Return information about this server."""
    return {
        "name": "handshake-demo-server",
        "version": "1.0.0"
    }


# =============================================================================
# Initialize Handler
# =============================================================================

def handle_initialize(request_id, params):
    """Handle the initialize request from a client.

    This is the first step of the MCP handshake. The client tells us:
    - What protocol version it supports
    - What capabilities it has
    - Information about itself

    We respond with:
    - The protocol version we'll use
    - Our capabilities
    - Information about ourselves
    """
    global connection

    # Check: Already initialized?
    if connection.state != ConnectionState.DISCONNECTED:
        return {
            "jsonrpc": "2.0",
            "id": request_id,
            "error": {
                "code": -32600,
                "message": "Invalid Request: Server already initialized"
            }
        }

    # Extract client information
    client_version = params.get("protocolVersion", "unknown")
    connection.client_info = params.get("clientInfo", {})
    connection.client_capabilities = params.get("capabilities", {})

    # Version negotiation
    # In a real server, you might check client_version and negotiate
    # For now, we just use our version
    connection.negotiated_version = MCPConnection.PROTOCOL_VERSION

    # Log the connection attempt
    client_name = connection.client_info.get("name", "unknown")
    client_ver = connection.client_info.get("version", "unknown")
    print(f"[INIT] Client connecting: {client_name} v{client_ver}")
    print(f"[INIT] Client protocol version: {client_version}")
    print(f"[INIT] Client capabilities: {list(connection.client_capabilities.keys())}")

    # Transition to INITIALIZING state
    connection.state = ConnectionState.INITIALIZING
    connection.connected_at = datetime.now()

    # Build and return the response
    return {
        "jsonrpc": "2.0",
        "id": request_id,
        "result": {
            "protocolVersion": connection.negotiated_version,
            "capabilities": get_server_capabilities(),
            "serverInfo": get_server_info()
        }
    }


# =============================================================================
# Initialized Notification Handler
# =============================================================================

def handle_initialized_notification():
    """Handle the initialized notification from the client.

    This notification signals that the client has processed our capabilities
    and is ready for normal operation. This completes the handshake.

    Important: This is a notification, not a request. We do NOT send a response.
    """
    global connection

    if connection.state == ConnectionState.DISCONNECTED:
        print("[WARN] Received initialized notification before initialize request")
        return None

    if connection.state == ConnectionState.READY:
        print("[WARN] Received duplicate initialized notification")
        return None

    # Transition to READY state - handshake complete!
    connection.state = ConnectionState.READY
    connection.ready_at = datetime.now()

    client_name = connection.client_info.get("name", "unknown")
    print(f"[READY] Handshake complete with {client_name}")
    print(f"[READY] Server capabilities: {list(get_server_capabilities().keys())}")
    print(f"[READY] Ready for normal operations")

    return None  # Notifications don't get responses


# =============================================================================
# Placeholder Handlers (for testing state validation)
# =============================================================================

def handle_tools_list(request_id, params):
    """Handle tools/list request - placeholder for Lesson 5."""
    return {
        "jsonrpc": "2.0",
        "id": request_id,
        "result": {
            "tools": [
                {
                    "name": "echo",
                    "description": "Echo back a message (demo tool)",
                    "inputSchema": {
                        "type": "object",
                        "properties": {
                            "message": {
                                "type": "string",
                                "description": "Message to echo"
                            }
                        },
                        "required": ["message"]
                    }
                }
            ]
        }
    }


def handle_tools_call(request_id, params):
    """Handle tools/call request - placeholder for Lesson 5."""
    tool_name = params.get("name")
    arguments = params.get("arguments", {})

    if tool_name == "echo":
        message = arguments.get("message", "")
        return {
            "jsonrpc": "2.0",
            "id": request_id,
            "result": {
                "content": [
                    {"type": "text", "text": f"Echo: {message}"}
                ]
            }
        }

    return {
        "jsonrpc": "2.0",
        "id": request_id,
        "error": {
            "code": -32602,
            "message": f"Unknown tool: {tool_name}"
        }
    }


# =============================================================================
# Request Router
# =============================================================================

def handle_request(request):
    """Route JSON-RPC requests to appropriate handlers.

    This is the main entry point for processing MCP messages.
    It enforces state-based access control.
    """
    method = request.get("method")
    request_id = request.get("id")  # None for notifications
    params = request.get("params", {})
    is_notification = request_id is None

    # Handle initialize - allowed in DISCONNECTED state only
    if method == "initialize":
        return handle_initialize(request_id, params)

    # Handle initialized notification
    if method == "notifications/initialized":
        return handle_initialized_notification()

    # All other methods require READY state
    if not connection.is_ready():
        if is_notification:
            # Can't respond to notifications, just log
            print(f"[WARN] Received {method} notification before ready")
            return None
        return {
            "jsonrpc": "2.0",
            "id": request_id,
            "error": {
                "code": -32002,
                "message": "Server not initialized. Send 'initialize' request first."
            }
        }

    # Route to appropriate handler
    handlers = {
        "tools/list": handle_tools_list,
        "tools/call": handle_tools_call,
        # Add more handlers as you implement more features
    }

    handler = handlers.get(method)
    if handler:
        return handler(request_id, params)

    # Method not found
    if is_notification:
        print(f"[WARN] Unknown notification: {method}")
        return None

    return {
        "jsonrpc": "2.0",
        "id": request_id,
        "error": {
            "code": -32601,
            "message": f"Method not found: {method}"
        }
    }


# =============================================================================
# HTTP Server
# =============================================================================

class MCPHandler(BaseHTTPRequestHandler):
    """HTTP request handler for MCP protocol."""

    def log_message(self, format, *args):
        """Custom logging format."""
        print(f"[HTTP] {args[0]}")

    def send_json(self, data, status=200):
        """Send a JSON response."""
        body = json.dumps(data, indent=2).encode("utf-8")
        self.send_response(status)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def do_POST(self):
        """Handle POST requests containing JSON-RPC messages."""
        # Read request body
        content_length = int(self.headers.get("Content-Length", 0))
        body = self.rfile.read(content_length)

        # Parse JSON
        try:
            request = json.loads(body.decode("utf-8"))
        except json.JSONDecodeError as e:
            self.send_json({
                "jsonrpc": "2.0",
                "id": None,
                "error": {
                    "code": -32700,
                    "message": f"Parse error: {e}"
                }
            }, status=400)
            return

        # Validate JSON-RPC structure
        if request.get("jsonrpc") != "2.0":
            self.send_json({
                "jsonrpc": "2.0",
                "id": request.get("id"),
                "error": {
                    "code": -32600,
                    "message": "Invalid Request: jsonrpc must be '2.0'"
                }
            }, status=400)
            return

        # Handle the request
        response = handle_request(request)

        # Notifications don't get responses
        if response is None:
            self.send_response(204)
            self.end_headers()
            return

        self.send_json(response)

    def do_GET(self):
        """Handle GET requests - return server status."""
        status = {
            "server": "MCP Handshake Demo Server",
            "state": connection.state.value,
            "connected_at": connection.connected_at.isoformat() if connection.connected_at else None,
            "ready_at": connection.ready_at.isoformat() if connection.ready_at else None,
            "client": connection.client_info
        }
        self.send_json(status)


# =============================================================================
# Server Entry Point
# =============================================================================

def run_server(host="localhost", port=8080):
    """Start the MCP server."""
    server = HTTPServer((host, port), MCPHandler)
    print("=" * 60)
    print("MCP Handshake Demo Server")
    print("=" * 60)
    print(f"Running on http://{host}:{port}")
    print(f"Protocol version: {MCPConnection.PROTOCOL_VERSION}")
    print(f"Capabilities: {list(get_server_capabilities().keys())}")
    print()
    print("GET  /        - Check server status")
    print("POST /        - Send JSON-RPC messages")
    print()
    print("Press Ctrl+C to stop")
    print("=" * 60)

    try:
        server.serve_forever()
    except KeyboardInterrupt:
        print("\nShutting down...")
        server.shutdown()


if __name__ == "__main__":
    run_server()

What You've Built

Congratulations! Your handshake server now correctly implements:

  • Connection state tracking — DISCONNECTED, INITIALIZING, and READY states
  • Initialize request handling — Extracts client info and capabilities
  • Server response — Returns protocol version, capabilities, and server info
  • Initialized notification — Properly handles the notification without a response
  • State-based access control — Rejects requests before initialization is complete
Production Considerations

In a production server, you would want to:

  • Handle multiple concurrent connections (connection per session)
  • Implement proper protocol version negotiation
  • Add timeout handling for incomplete handshakes
  • Log handshake events for debugging
