Building the Server

Build a complete MCP server with proper handshake handling, step by step.

Create: mcp_server.py

Let's build your MCP server piece by piece. Create a new file called mcp_server.py and add each section as we go.

Section 1: Imports and Connection State

Start with the imports and connection state management:

python mcp_server.py 
"""
mcp_server.py - MCP server with proper handshake handling.

Run: python mcp_server.py
Test: curl commands below
"""

import json
from http.server import HTTPServer, BaseHTTPRequestHandler
from enum import Enum
from typing import Any, cast


# =============================================================================
# Connection State
# =============================================================================

class ConnectionState(Enum):
    """Track the state of an MCP connection."""
    DISCONNECTED = "disconnected"
    INITIALIZING = "initializing"
    READY = "ready"


# Server configuration
PROTOCOL_VERSION = "2024-11-05"
SERVER_NAME = "mcp-handshake-server"
SERVER_VERSION = "1.0.0"

# Connection state (global for simplicity)
state = ConnectionState.DISCONNECTED
client_info: dict[str, Any] = {}
client_capabilities: dict[str, Any] = {}

Section 3: Initialize Handler

Add this after the server capabilities:

python mcp_server.py - Add after Section 2 
# =============================================================================
# Initialize Handler
# =============================================================================

def handle_initialize(request_id: int | str, params: dict) -> dict:
    """Handle the initialize request from a client."""
    global state, client_info, client_capabilities

    # Already initialized?
    if state != ConnectionState.DISCONNECTED:
        return {
            "jsonrpc": "2.0",
            "id": request_id,
            "error": {
                "code": -32600,
                "message": "Already initialized"
            }
        }

    # Store client information
    client_info = params.get("clientInfo", {})
    client_capabilities = params.get("capabilities", {})

    # Log the connection
    name = client_info.get("name", "unknown")
    version = client_info.get("version", "unknown")
    print(f"[INIT] Client: {name} v{version}")

    # Transition to INITIALIZING
    state = ConnectionState.INITIALIZING

    # Return our capabilities
    return {
        "jsonrpc": "2.0",
        "id": request_id,
        "result": {
            "protocolVersion": PROTOCOL_VERSION,
            "capabilities": get_server_capabilities(),
            "serverInfo": get_server_info()
        }
    }

# =============================================================================
# Initialized Notification Handler
# =============================================================================

def handle_initialized_notification() -> None:
    """Handle the initialized notification from the client."""
    global state

    if state == ConnectionState.INITIALIZING:
        state = ConnectionState.READY
        print(f"[READY] Handshake complete!")
    else:
        print(f"[WARN] Unexpected initialized notification in state: {state}")

                }

def handle_tools_call(request_id: int | str, params: dict) -> dict:
    """Execute a tool."""
    name = params.get("name")
    args = params.get("arguments", {})

    if name == "echo":
        return {
            "jsonrpc": "2.0",
            "id": request_id,
            "result": {
                "content": [{"type": "text", "text": args.get("message", "")}]
            }
        }

    return {
        "jsonrpc": "2.0",
        "id": request_id,
        "error": {"code": -32602, "message": f"Unknown tool: {name}"}
    }

Section 6: Request Router

Add the main request router:

python mcp_server.py - Add after Section 5 
# =============================================================================
# Request Router
# =============================================================================

def handle_request(request: dict) -> dict | None:
    """Route JSON-RPC requests to appropriate handlers."""
    method = request.get("method")
    request_id = cast(int | str | None, request.get("id"))
    params = request.get("params", {})
    is_notification = request_id is None

    # Initialize - allowed before ready
    if method == "initialize":
        return handle_initialize(request_id, params)

    # Initialized notification
    if method == "notifications/initialized":
        handle_initialized_notification()
        return None  # No response for notifications

    # All other methods require READY state
    if state != ConnectionState.READY:
        if is_notification:
            print(f"[WARN] {method} received before ready")
            return None
        return {
            "jsonrpc": "2.0",
            "id": request_id,
            "error": {
                "code": -32002,
                "message": "Server not initialized"
            }
        }

    # Route to handlers
    if method == "tools/list":
        return handle_tools_list(request_id)
    elif method == "tools/call":
        return handle_tools_call(request_id, params)

    # Unknown method
    if is_notification:
        return None
    return {
        "jsonrpc": "2.0",
        "id": request_id,
        "error": {"code": -32601, "message": f"Method not found: {method}"}
    }

Section 7: HTTP Server

Add the HTTP server at the end:

python mcp_server.py - Add after Section 6 
# =============================================================================
# HTTP Server
# =============================================================================

class MCPHandler(BaseHTTPRequestHandler):
    """HTTP handler for MCP requests."""

    def log_message(self, format: str, *args) -> None:
        print(f"[HTTP] {args[0]}")

    def send_json(self, data: dict, status: int = 200) -> None:
        body = json.dumps(data, indent=2).encode()
        self.send_response(status)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def do_GET(self) -> None:
        """Return server status."""
        self.send_json({
            "server": SERVER_NAME,
            "version": SERVER_VERSION,
            "state": state.value,
            "client": client_info or None
        })

    def do_POST(self) -> None:
        """Handle JSON-RPC requests."""
        length = int(self.headers.get("Content-Length", 0))
        body = self.rfile.read(length)

        try:
            request = json.loads(body)
        except json.JSONDecodeError as e:
            self.send_json({
                "jsonrpc": "2.0",
                "id": None,
                "error": {"code": -32700, "message": f"Parse error: {e}"}
            }, 400)
            return

        if request.get("jsonrpc") != "2.0":
            self.send_json({
                "jsonrpc": "2.0",
                "id": request.get("id"),
                "error": {"code": -32600, "message": "jsonrpc must be '2.0'"}
            }, 400)
            return

        response = handle_request(request)

        if response is None:
            self.send_response(204)
            self.end_headers()
        else:
            self.send_json(response)


def run_server(host: str = "localhost", port: int = 8080) -> None:
    """Start the MCP server."""
    server = HTTPServer((host, port), MCPHandler)
    print(f"MCP Server running on http://{host}:{port}")
    print("Press Ctrl+C to stop\n")
    try:
        server.serve_forever()
    except KeyboardInterrupt:
        print("\nShutting down...")


if __name__ == "__main__":
    run_server()

What You've Built

Your MCP server now properly implements:

  • State tracking — DISCONNECTED → INITIALIZING → READY
  • Initialize request — Stores client info, returns capabilities
  • Initialized notification — Completes handshake, no response sent
  • Access control — Rejects requests before initialization
