Section 1: Imports and Connection State

Create a new file called mcp_core.py with these imports:

python mcp_core.py 
"""
mcp_core.py - MCP server with proper handshake handling.

This module builds on jsonrpc.py from Lesson 3.

Run: python mcp_core.py
Test: curl commands below
"""

import json
import sys
from http.server import HTTPServer, BaseHTTPRequestHandler
from enum import Enum
from typing import Any

# Import our JSON-RPC utilities from Lesson 3
from jsonrpc import (
    JsonRpcRouter,
    JsonRpcError,
    JsonRpcErrorCode,
    create_error_response,
)


# =============================================================================
# Connection State
# =============================================================================

class ConnectionState(Enum):
    """Track the state of an MCP connection."""
    DISCONNECTED = "disconnected"
    INITIALIZING = "initializing"
    READY = "ready"


# Server configuration
PROTOCOL_VERSION = "2024-11-05"
SERVER_NAME = "mcp-handshake-server"
SERVER_VERSION = "1.0.0"

# Connection state (global for simplicity)
state = ConnectionState.DISCONNECTED
client_info: dict[str, Any] = {}
client_capabilities: dict[str, Any] = {}

# Create our router using JsonRpcRouter from Lesson 3
router = JsonRpcRouter()

Section 3: MCP Handshake Handlers

Add the handshake handlers after Section 2. Notice how we use the @router.method() decorator from Lesson 3 to register handlers:

python mcp_core.py - Add after Section 2 
# =============================================================================
# MCP Handshake Handlers
# =============================================================================

@router.method("initialize")
def handle_initialize(params: dict) -> dict:
    """Handle the initialize request from a client."""
    global state, client_info, client_capabilities

    # Already initialized?
    if state != ConnectionState.DISCONNECTED:
        raise JsonRpcError(
            JsonRpcErrorCode.INVALID_REQUEST,
            "Already initialized"
        )

    # Store client information
    client_info = params.get("clientInfo", {})
    client_capabilities = params.get("capabilities", {})

    # Log the connection (stderr only for STDIO compatibility)
    name = client_info.get("name", "unknown")
    version = client_info.get("version", "unknown")
    print(f"[INIT] Client: {name} v{version}", file=sys.stderr)

    # Transition to INITIALIZING
    state = ConnectionState.INITIALIZING

    # Return our capabilities (router wraps this in a success response)
    return {
        "protocolVersion": PROTOCOL_VERSION,
        "capabilities": get_server_capabilities(),
        "serverInfo": get_server_info()
    }


@router.notification("notifications/initialized")
def handle_initialized(params: dict) -> None:
    """Handle the initialized notification from the client."""
    global state

    if state == ConnectionState.INITIALIZING:
        state = ConnectionState.READY
        print("[READY] Handshake complete!", file=sys.stderr)
    else:
        print(f"[WARN] Unexpected initialized notification in state: {state}", file=sys.stderr)
The Decorator Pattern

Using @router.method("initialize") automatically registers the handler. The router calls it when a matching request arrives, and wraps the return value in a proper JSON-RPC response. No manual response building needed!

Section 5: State-Aware Request Handler

Add a wrapper that enforces the MCP state machine before delegating to the router:

python mcp_core.py - Add after Section 4 
# =============================================================================
# State-Aware Request Handler
# =============================================================================

def handle_request(request: dict) -> dict | None:
    """
    Process a JSON-RPC request with MCP state checking.

    The router handles method dispatch and response formatting.
    We just need to enforce the state machine.
    """
    method = request.get("method", "")
    request_id = request.get("id")

    # Initialize is always allowed
    if method == "initialize":
        return router.handle(request)

    # Initialized notification is always allowed
    if method == "notifications/initialized":
        return router.handle(request)

    # All other methods require READY state
    if state != ConnectionState.READY:
        if request_id is None:  # Notification
            print(f"[WARN] {method} received before ready", file=sys.stderr)
            return None
        return create_error_response(
            request_id=request_id,
            code=JsonRpcErrorCode.INVALID_REQUEST,
            message="Server not initialized"
        )

    # Delegate to the router for everything else
    return router.handle(request)

Section 6: HTTP Server

Add the HTTP server at the end:

python mcp_core.py - Add after Section 5 
# =============================================================================
# HTTP Server
# =============================================================================

class MCPHandler(BaseHTTPRequestHandler):
    """HTTP handler for MCP requests."""

    def log_message(self, format: str, *args) -> None:
        print(f"[HTTP] {args[0]}")

    def send_json(self, data: dict, status: int = 200) -> None:
        body = json.dumps(data, indent=2).encode()
        self.send_response(status)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def do_GET(self) -> None:
        """Return server status."""
        self.send_json({
            "server": SERVER_NAME,
            "version": SERVER_VERSION,
            "state": state.value,
            "client": client_info or None
        })

    def do_POST(self) -> None:
        """Handle JSON-RPC requests."""
        length = int(self.headers.get("Content-Length", 0))
        body = self.rfile.read(length)

        try:
            request = json.loads(body)
        except json.JSONDecodeError as e:
            self.send_json(create_error_response(
                request_id=None,
                code=JsonRpcErrorCode.PARSE_ERROR,
                message=f"Parse error: {e}"
            ), 400)
            return

        if request.get("jsonrpc") != "2.0":
            self.send_json(create_error_response(
                request_id=request.get("id"),
                code=JsonRpcErrorCode.INVALID_REQUEST,
                message="jsonrpc must be '2.0'"
            ), 400)
            return

        response = handle_request(request)

        if response is None:
            self.send_response(204)
            self.end_headers()
        else:
            self.send_json(response)


def run_server(host: str = "localhost", port: int = 8080) -> None:
    """Start the MCP server."""
    server = HTTPServer((host, port), MCPHandler)
    print(f"MCP Server running on http://{host}:{port}")
    print("Press Ctrl+C to stop\n")
    try:
        server.serve_forever()
    except KeyboardInterrupt:
        print("\nShutting down...")


if __name__ == "__main__":
    run_server()

What You've Built

Your MCP server now properly implements:

  • Module reuse — Imports JsonRpcRouter and helpers from Lesson 3
  • State tracking — DISCONNECTED → INITIALIZING → READY
  • Decorator-based handlers — Clean @router.method() syntax
  • Initialize request — Stores client info, returns capabilities
  • Initialized notification — Completes handshake, no response sent
  • Access control — Rejects requests before initialization
