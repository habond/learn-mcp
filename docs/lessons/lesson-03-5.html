Building a Router

Create a method router that dispatches JSON-RPC messages to handlers.

What is a Method Router?

A method router maps incoming JSON-RPC method names to handler functions. It's the central dispatch mechanism for your server, routing messages like tools/list to the appropriate handler.

text 
Incoming Request                   Router                      Handler
+-------------------+          +------------+          +------------------+
| method: tools/list| -------> |   Match    | -------> | handle_tools_list|
+-------------------+          |   method   |          +------------------+
                               +------------+
                                     |
+-------------------+                |
| method: tools/call| ---------------+-------> handle_tools_call
+-------------------+                |
                                     |
+-------------------+                |
| method: initialize| ---------------+-------> handle_initialize
+-------------------+

Basic Router Implementation

Let's start with a simple router that maps methods to handler functions:

python 
"""
jsonrpc_router.py - A simple JSON-RPC 2.0 method router.
"""

from typing import Callable, Any, Optional
from enum import IntEnum


class JsonRpcErrorCode(IntEnum):
    PARSE_ERROR = -32700
    INVALID_REQUEST = -32600
    METHOD_NOT_FOUND = -32601
    INVALID_PARAMS = -32602
    INTERNAL_ERROR = -32603


class JsonRpcError(Exception):
    """JSON-RPC error that can be converted to an error response."""

    def __init__(self, code: JsonRpcErrorCode, message: str, data: Any = None):
        self.code = code
        self.message = message
        self.data = data
        super().__init__(message)


class JsonRpcRouter:
    """
    Routes JSON-RPC methods to handler functions.

    Usage:
        router = JsonRpcRouter()

        @router.method("tools/list")
        def handle_tools_list(params):
            return {"tools": [...]}

        response = router.handle(request)
    """

    def __init__(self):
        self._request_handlers: dict[str, Callable] = {}
        self._notification_handlers: dict[str, Callable] = {}

    def method(self, name: str):
        """Decorator to register a request handler."""
        def decorator(func: Callable):
            self._request_handlers[name] = func
            return func
        return decorator

    def notification(self, name: str):
        """Decorator to register a notification handler."""
        def decorator(func: Callable):
            self._notification_handlers[name] = func
            return func
        return decorator

    def handle(self, message: dict) -> Optional[dict]:
        """
        Process a JSON-RPC message and return response if needed.

        Args:
            message: Parsed JSON-RPC message

        Returns:
            Response dict for requests, None for notifications
        """
        # Detect message type
        is_request = "id" in message and "method" in message
        is_notification = "id" not in message and "method" in message

        if is_notification:
            self._handle_notification(message)
            return None

        if is_request:
            return self._handle_request(message)

        # Invalid message
        return self._error_response(
            request_id=message.get("id"),
            code=JsonRpcErrorCode.INVALID_REQUEST,
            message="Invalid JSON-RPC message"
        )

    def _handle_request(self, request: dict) -> dict:
        """Handle a request and return a response."""
        request_id = request.get("id")
        method = request.get("method")
        params = request.get("params", {})

        # Find handler
        handler = self._request_handlers.get(method)
        if not handler:
            return self._error_response(
                request_id=request_id,
                code=JsonRpcErrorCode.METHOD_NOT_FOUND,
                message=f"Method not found: {method}",
                data={"method": method}
            )

        # Execute handler
        try:
            result = handler(params)
            return {
                "jsonrpc": "2.0",
                "id": request_id,
                "result": result
            }
        except JsonRpcError as e:
            return self._error_response(
                request_id=request_id,
                code=e.code,
                message=e.message,
                data=e.data
            )
        except Exception as e:
            return self._error_response(
                request_id=request_id,
                code=JsonRpcErrorCode.INTERNAL_ERROR,
                message=str(e)
            )

    def _handle_notification(self, notification: dict):
        """Handle a notification (no response)."""
        method = notification.get("method")
        params = notification.get("params", {})

        handler = self._notification_handlers.get(method)
        if handler:
            try:
                handler(params)
            except Exception as e:
                # Log but don't respond
                print(f"Notification error ({method}): {e}")

    def _error_response(
        self,
        request_id: Any,
        code: JsonRpcErrorCode,
        message: str,
        data: Any = None
    ) -> dict:
        """Create an error response."""
        error = {"code": int(code), "message": message}
        if data is not None:
            error["data"] = data

        return {
            "jsonrpc": "2.0",
            "id": request_id,
            "error": error
        }

Using the Router

Register handlers using decorators for a clean API:

python 
"""Example usage of the JSON-RPC router."""

router = JsonRpcRouter()


@router.method("initialize")
def handle_initialize(params: dict) -> dict:
    """Handle the initialize request."""
    client_info = params.get("clientInfo", {})
    print(f"Client connected: {client_info.get('name', 'unknown')}")

    return {
        "protocolVersion": "2024-11-05",
        "capabilities": {
            "tools": {},
            "resources": {}
        },
        "serverInfo": {
            "name": "example-server",
            "version": "1.0.0"
        }
    }


@router.method("tools/list")
def handle_tools_list(params: dict) -> dict:
    """Return list of available tools."""
    return {
        "tools": [
            {
                "name": "calculator",
                "description": "Performs basic math operations",
                "inputSchema": {
                    "type": "object",
                    "properties": {
                        "expression": {
                            "type": "string",
                            "description": "Math expression to evaluate"
                        }
                    },
                    "required": ["expression"]
                }
            }
        ]
    }


@router.method("tools/call")
def handle_tools_call(params: dict) -> dict:
    """Execute a tool."""
    tool_name = params.get("name")
    arguments = params.get("arguments", {})

    if tool_name == "calculator":
        expression = arguments.get("expression", "")
        try:
            # WARNING: eval is dangerous! Use a safe parser in production
            result = eval(expression)
            return {
                "content": [
                    {"type": "text", "text": f"Result: {result}"}
                ]
            }
        except Exception as e:
            raise JsonRpcError(
                JsonRpcErrorCode.INVALID_PARAMS,
                f"Invalid expression: {e}"
            )

    raise JsonRpcError(
        JsonRpcErrorCode.INVALID_PARAMS,
        f"Unknown tool: {tool_name}",
        {"tool": tool_name}
    )


@router.notification("notifications/initialized")
def handle_initialized(params: dict):
    """Client has finished initializing."""
    print("Client initialization complete")


@router.notification("notifications/cancelled")
def handle_cancelled(params: dict):
    """Client cancelled a request."""
    request_id = params.get("requestId")
    print(f"Request {request_id} was cancelled")

Complete Server with Router

Here's a complete HTTP server using our JSON-RPC router:

python 
"""
jsonrpc_server.py - Complete JSON-RPC server with routing.
"""

import json
from http.server import HTTPServer, BaseHTTPRequestHandler


class JsonRpcHandler(BaseHTTPRequestHandler):
    """HTTP handler that routes JSON-RPC requests."""

    # Class-level router (shared by all handlers)
    router: JsonRpcRouter = None

    def log_message(self, format, *args):
        """Custom logging."""
        print(f"[{self.log_date_time_string()}] {format % args}")

    def do_POST(self):
        """Handle POST requests containing JSON-RPC messages."""
        # Validate content type
        content_type = self.headers.get("Content-Type", "")
        if "application/json" not in content_type:
            self._send_error(415, "Content-Type must be application/json")
            return

        # Read body
        content_length = int(self.headers.get("Content-Length", 0))
        if content_length == 0:
            self._send_error(400, "Empty request body")
            return

        body = self.rfile.read(content_length)

        # Parse JSON
        try:
            message = json.loads(body.decode("utf-8"))
        except json.JSONDecodeError as e:
            response = {
                "jsonrpc": "2.0",
                "id": None,
                "error": {
                    "code": -32700,
                    "message": f"Parse error: {e}"
                }
            }
            self._send_json(response, 200)
            return

        # Validate basic structure
        if not isinstance(message, dict):
            response = {
                "jsonrpc": "2.0",
                "id": None,
                "error": {
                    "code": -32600,
                    "message": "Request must be a JSON object"
                }
            }
            self._send_json(response, 200)
            return

        if message.get("jsonrpc") != "2.0":
            response = {
                "jsonrpc": "2.0",
                "id": message.get("id"),
                "error": {
                    "code": -32600,
                    "message": "jsonrpc must be '2.0'"
                }
            }
            self._send_json(response, 200)
            return

        # Route the message
        response = self.router.handle(message)

        # Send response (None for notifications)
        if response is not None:
            self._send_json(response, 200)
        else:
            # No content for notifications
            self.send_response(204)
            self.end_headers()

    def _send_json(self, data: dict, status: int = 200):
        """Send a JSON response."""
        body = json.dumps(data, indent=2).encode("utf-8")
        self.send_response(status)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def _send_error(self, status: int, message: str):
        """Send an HTTP error response."""
        body = json.dumps({"error": message}).encode("utf-8")
        self.send_response(status)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)


def run_server(router: JsonRpcRouter, host: str = "localhost", port: int = 8080):
    """Start the JSON-RPC server."""
    JsonRpcHandler.router = router
    server = HTTPServer((host, port), JsonRpcHandler)

    print(f"JSON-RPC server running on http://{host}:{port}")
    print("Press Ctrl+C to stop")

    try:
        server.serve_forever()
    except KeyboardInterrupt:
        print("\nShutting down...")
        server.shutdown()


if __name__ == "__main__":
    # Create router and register handlers
    router = JsonRpcRouter()

    @router.method("initialize")
    def handle_initialize(params):
        return {
            "protocolVersion": "2024-11-05",
            "capabilities": {"tools": {}},
            "serverInfo": {"name": "demo-server", "version": "1.0.0"}
        }

    @router.method("tools/list")
    def handle_tools_list(params):
        return {"tools": []}

    @router.notification("notifications/initialized")
    def on_initialized(params):
        print("Client ready!")

    # Start server
    run_server(router)

Testing the Server

Test your server with curl:

bash 
# Start the server
python jsonrpc_server.py

# In another terminal, test initialize
curl -X POST http://localhost:8080 \
  -H "Content-Type: application/json" \
  -d '{
    "jsonrpc": "2.0",
    "id": 1,
    "method": "initialize",
    "params": {
      "protocolVersion": "2024-11-05",
      "clientInfo": {"name": "test-client", "version": "1.0"}
    }
  }'

# Response:

#     "serverInfo": {"name": "demo-server", "version": "1.0.0"}

# }
