Complete STDIO Server

Build a fully functional MCP server using STDIO transport.

Building Incrementally

Let's build a complete STDIO MCP server step by step. We'll create stdio_server.py incrementally, testing as we go.

File Structure

We'll build one file: stdio_server.py. Each section below adds to this file. Follow along and add each section in order.

Section 1: Imports and Logging Setup

Create: stdio_server.py

Start with the imports and logging (to stderr only!):

python stdio_server.py 
#!/usr/bin/env python3
"""
stdio_server.py - MCP server using STDIO transport.

Run with: python stdio_server.py
Test with: echo commands piped to stdin
"""

import sys
import json
import logging
from datetime import datetime
from typing import cast


# =============================================================================
# Section 1: Logging Setup (stderr only!)
# =============================================================================

logging.basicConfig(
    level=logging.DEBUG,
    format='%(asctime)s [%(levelname)s] %(message)s',
    handlers=[logging.StreamHandler(sys.stderr)]  # CRITICAL: stderr only!
)
logger = logging.getLogger("mcp-stdio-server")


class ErrorCode:
    """JSON-RPC error codes."""
    PARSE_ERROR = -32700
    INVALID_REQUEST = -32600
    METHOD_NOT_FOUND = -32601
    INVALID_PARAMS = -32602
    INTERNAL_ERROR = -32603


# Quick test
if __name__ == "__main__":
    logger.info("Logging test - this goes to stderr")
    print("Print test - this goes to stdout")

Try It

bash 
python stdio_server.py

Expected Output

You'll see both outputs - logging to stderr and print to stdout.

Section 2: Response Functions

Edit: stdio_server.py

Add the following after Section 1:

python 
# =============================================================================
# Section 2: Response Functions (stdout)
# =============================================================================

def write_response(response: dict) -> None:
    """Write a JSON-RPC response to stdout."""
    # Use compact JSON - no pretty printing for STDIO!
    json_str = json.dumps(response, separators=(',', ':'))
    sys.stdout.write(json_str + "\n")
    sys.stdout.flush()  # CRITICAL: Always flush!
    logger.debug(f"Sent response for id={response.get('id')}")


def send_result(msg_id: int, result: dict) -> None:
    """Send a successful JSON-RPC response."""
    write_response({
        "jsonrpc": "2.0",
        "id": msg_id,
        "result": result
    })


def send_error(msg_id: int | None, code: int, message: str) -> None:
    """Send a JSON-RPC error response."""
    write_response({
        "jsonrpc": "2.0",
        "id": msg_id,
        "error": {"code": code, "message": message}
    })

TOOLS = [
    {
        "name": "echo",
        "description": "Echo back a message. Useful for testing.",
        "inputSchema": {
            "type": "object",
            "properties": {
                "message": {
                    "type": "string",
                    "description": "The message to echo back"
                }
            },
            "required": ["message"]
        }
    },
    {
        "name": "get_time",
        "description": "Get the current server time.",
        "inputSchema": {
            "type": "object",
            "properties": {},
            "required": []
        }
    },
    {
        "name": "add_numbers",
        "description": "Add two numbers together.",
        "inputSchema": {
            "type": "object",
            "properties": {
                "a": {"type": "number", "description": "First number"},
                "b": {"type": "number", "description": "Second number"}
            },
            "required": ["a", "b"]
        }
    }
]


def execute_tool(name: str, arguments: dict) -> dict:
    """Execute a tool and return the result content."""
    logger.info(f"Executing tool: {name}")

    if name == "echo":
        message = arguments.get("message", "")
        return {"type": "text", "text": message}

    elif name == "get_time":
        current_time = datetime.now().isoformat()
        return {"type": "text", "text": f"Current time: {current_time}"}

    elif name == "add_numbers":
        a = arguments.get("a", 0)
        b = arguments.get("b", 0)
        return {"type": "text", "text": f"{a} + {b} = {a + b}"}

    else:
        raise ValueError(f"Unknown tool: {name}")

# =============================================================================
# Section 4: MCP Method Handlers
# =============================================================================

def handle_initialize(msg_id: int | str, params: dict) -> None:
    """Handle the initialize request."""
    client_info = params.get("clientInfo", {})
    logger.info(f"Client connecting: {client_info.get('name', 'unknown')}")

    send_result(msg_id, {
        "protocolVersion": "2024-11-05",
        "capabilities": {"tools": {}},
        "serverInfo": {
            "name": "stdio-demo-server",
            "version": "1.0.0"
        }
    })


def handle_tools_list(msg_id: int | str, params: dict) -> None:
    """Handle the tools/list request."""
    logger.debug("Listing tools")
    send_result(msg_id, {"tools": TOOLS})


def handle_tools_call(msg_id: int | str, params: dict) -> None:
    """Handle the tools/call request."""
    tool_name = params.get("name")
    arguments = params.get("arguments", {})

    try:
        result = execute_tool(tool_name, arguments)
        send_result(msg_id, {"content": [result]})
    except ValueError as e:
        send_error(msg_id, ErrorCode.METHOD_NOT_FOUND, str(e))
    except Exception as e:
        logger.error(f"Tool error: {e}")
        send_error(msg_id, ErrorCode.INTERNAL_ERROR, str(e))

Section 5: Message Router and Main Loop

Edit: stdio_server.py

Add the following after Section 4:

python 
# =============================================================================
# Section 5: Message Router and Main Loop
# =============================================================================

def handle_message(message: dict) -> None:
    """Route a JSON-RPC message to the appropriate handler."""
    if message.get("jsonrpc") != "2.0":
        logger.error("Invalid JSON-RPC version")
        return

    method = message.get("method")
    msg_id = cast(int | str | None, message.get("id"))
    params = message.get("params", {})

    # Notifications (no id) - no response needed
    if method and msg_id is None:
        if method == "notifications/initialized":
            logger.info("Client initialization complete")
        else:
            logger.warning(f"Unknown notification: {method}")
        return

    # Requests (has method and id)
    if method and msg_id is not None:
        logger.debug(f"Request [{msg_id}]: {method}")

        handlers = {
            "initialize": handle_initialize,
            "tools/list": handle_tools_list,
            "tools/call": handle_tools_call,
        }

        handler = handlers.get(method)
        if handler:
            handler(msg_id, params)
        else:
            send_error(msg_id, ErrorCode.METHOD_NOT_FOUND, f"Unknown method: {method}")


def main() -> None:
    """Main server loop - read and process messages until EOF."""
    logger.info("STDIO MCP Server starting...")

    while True:
        try:
            # Read one line from stdin
            line = sys.stdin.readline()

            # EOF - client disconnected
            if not line:
                logger.info("EOF received, shutting down")
                break

            # Skip empty lines
            line = line.strip()
            if not line:
                continue

            # Parse JSON
            try:
                message = json.loads(line)
            except json.JSONDecodeError as e:
                logger.error(f"JSON parse error: {e}")
                continue

            # Handle the message
            handle_message(message)

        except KeyboardInterrupt:
            logger.info("Interrupted, shutting down")
            break
        except Exception as e:
            logger.error(f"Unexpected error: {e}", exc_info=True)

    logger.info("Server shutdown complete")

Replace the if __name__ == "__main__" block:

python 
if __name__ == "__main__":
    main()

What You've Built

Your STDIO MCP server now:

  • Reads JSON-RPC messages from stdin (one per line)
  • Writes compact JSON responses to stdout
  • Logs debug information to stderr
  • Handles the MCP initialization handshake
  • Exposes three tools: echo, get_time, add_numbers
  • Handles errors gracefully
Common Mistakes to Avoid
  • Using print() anywhere (it goes to stdout!)
  • Forgetting to call flush() after writing
  • Pretty-printing JSON responses (breaks line-based protocol)
  • Not handling EOF properly (server hangs)
