Complete STDIO Server

Build a fully functional MCP server using STDIO transport.

Putting It All Together

Now let's combine everything we've learned into a complete, production-ready MCP server using STDIO transport. This server will:

  • Handle the MCP initialization handshake
  • Expose a simple tool for demonstration
  • Use proper stdin/stdout/stderr separation
  • Include comprehensive logging

The Complete Server

Save this as stdio_server.py:

python 
#!/usr/bin/env python3
"""
Complete MCP Server using STDIO Transport.

This server demonstrates proper STDIO handling for MCP:
- stdin: JSON-RPC requests from client
- stdout: JSON-RPC responses to client
- stderr: Debug logging

Usage:
    python stdio_server.py

Or with Claude Desktop configuration:
    {
        "mcpServers": {
            "my-server": {
                "command": "python",
                "args": ["/path/to/stdio_server.py"]
            }
        }
    }
"""

import sys
import json
import logging
from datetime import datetime

# ============================================================================
# Logging Setup (stderr only!)
# ============================================================================

logging.basicConfig(
    level=logging.DEBUG,
    format='%(asctime)s [%(levelname)s] %(message)s',
    handlers=[logging.StreamHandler(sys.stderr)]
)
logger = logging.getLogger("mcp-stdio-server")

# ============================================================================
# JSON-RPC Error Codes
# ============================================================================

class ErrorCode:
    PARSE_ERROR = -32700
    INVALID_REQUEST = -32600
    METHOD_NOT_FOUND = -32601
    INVALID_PARAMS = -32602
    INTERNAL_ERROR = -32603

# ============================================================================
# Response Functions (stdout)
# ============================================================================

def write_response(response):
    """Write a JSON-RPC response to stdout."""
    json_str = json.dumps(response, separators=(',', ':'))
    sys.stdout.write(json_str + "\n")
    sys.stdout.flush()
    logger.debug(f"Sent: {response.get('id', 'notification')}")

def send_result(msg_id, result):
    """Send a successful JSON-RPC response."""
    write_response({
        "jsonrpc": "2.0",
        "id": msg_id,
        "result": result
    })

def send_error(msg_id, code, message, data=None):
    """Send a JSON-RPC error response."""
    error = {"code": code, "message": message}
    if data:
        error["data"] = data
    write_response({
        "jsonrpc": "2.0",
        "id": msg_id,
        "error": error
    })

# ============================================================================
# Tool Definitions
# ============================================================================

TOOLS = [
    {
        "name": "echo",
        "description": "Echo back a message. Useful for testing.",
        "inputSchema": {
            "type": "object",
            "properties": {
                "message": {
                    "type": "string",
                    "description": "The message to echo back"
                }
            },
            "required": ["message"]
        }
    },
    {
        "name": "get_time",
        "description": "Get the current server time.",
        "inputSchema": {
            "type": "object",
            "properties": {},
            "required": []
        }
    },
    {
        "name": "add_numbers",
        "description": "Add two numbers together.",
        "inputSchema": {
            "type": "object",
            "properties": {
                "a": {
                    "type": "number",
                    "description": "First number"
                },
                "b": {
                    "type": "number",
                    "description": "Second number"
                }
            },
            "required": ["a", "b"]
        }
    }
]

def execute_tool(name, arguments):
    """Execute a tool and return the result."""
    logger.info(f"Executing tool: {name} with args: {arguments}")

    if name == "echo":
        message = arguments.get("message", "")
        return {"type": "text", "text": message}

    elif name == "get_time":
        current_time = datetime.now().isoformat()
        return {"type": "text", "text": f"Current time: {current_time}"}

    elif name == "add_numbers":
        a = arguments.get("a", 0)
        b = arguments.get("b", 0)
        result = a + b
        return {"type": "text", "text": f"{a} + {b} = {result}"}

    else:
        raise ValueError(f"Unknown tool: {name}")

# ============================================================================
# MCP Method Handlers
# ============================================================================

def handle_initialize(msg_id, params):
    """Handle the initialize request."""
    client_info = params.get("clientInfo", {})
    logger.info(f"Client connecting: {client_info.get('name', 'unknown')}")

    send_result(msg_id, {
        "protocolVersion": "2024-11-05",
        "capabilities": {
            "tools": {}
        },
        "serverInfo": {
            "name": "stdio-demo-server",
            "version": "1.0.0"
        }
    })

def handle_tools_list(msg_id, params):
    """Handle the tools/list request."""
    logger.debug("Listing tools")
    send_result(msg_id, {"tools": TOOLS})

def handle_tools_call(msg_id, params):
    """Handle the tools/call request."""
    tool_name = params.get("name")
    arguments = params.get("arguments", {})

    try:
        result = execute_tool(tool_name, arguments)
        send_result(msg_id, {
            "content": [result]
        })
    except ValueError as e:
        send_error(msg_id, ErrorCode.METHOD_NOT_FOUND, str(e))
    except Exception as e:
        logger.error(f"Tool execution error: {e}")
        send_error(msg_id, ErrorCode.INTERNAL_ERROR, str(e))

def handle_notification(method, params):
    """Handle notifications (no response expected)."""
    if method == "notifications/initialized":
        logger.info("Client initialization complete")
    elif method == "notifications/cancelled":
        logger.info(f"Request cancelled: {params}")
    else:
        logger.warning(f"Unknown notification: {method}")

# ============================================================================
# Message Router
# ============================================================================

def handle_message(message):
    """Route a JSON-RPC message to the appropriate handler."""

    # Validate JSON-RPC version
    if message.get("jsonrpc") != "2.0":
        logger.error("Invalid JSON-RPC version")
        return

    method = message.get("method")
    msg_id = message.get("id")
    params = message.get("params", {})

    # Notifications (no id)
    if method and msg_id is None:
        handle_notification(method, params)
        return

    # Requests (has method and id)
    if method and msg_id is not None:
        logger.debug(f"Request [{msg_id}]: {method}")

        handlers = {
            "initialize": handle_initialize,
            "tools/list": handle_tools_list,
            "tools/call": handle_tools_call,
        }

        handler = handlers.get(method)
        if handler:
            handler(msg_id, params)
        else:
            send_error(msg_id, ErrorCode.METHOD_NOT_FOUND,
                      f"Method not found: {method}")
        return

    # Unknown message type
    logger.warning(f"Unknown message format: {message}")

# ============================================================================
# Main Loop
# ============================================================================

def main():
    """Main server loop - read and process messages until EOF."""
    logger.info("STDIO MCP Server starting...")

    while True:
        try:
            # Read one line from stdin
            line = sys.stdin.readline()

            # EOF - client disconnected
            if not line:
                logger.info("EOF received, shutting down")
                break

            # Skip empty lines
            line = line.strip()
            if not line:
                continue

            # Parse JSON
            try:
                message = json.loads(line)
            except json.JSONDecodeError as e:
                logger.error(f"JSON parse error: {e}")
                logger.error(f"Raw input: {repr(line)}")
                continue

            # Handle the message
            handle_message(message)

        except KeyboardInterrupt:
            logger.info("Interrupted, shutting down")
            break
        except Exception as e:
            logger.error(f"Unexpected error: {e}", exc_info=True)

    logger.info("Server shutdown complete")

if __name__ == "__main__":
    main()

Testing the Server

You can test the server by sending JSON-RPC messages through stdin:

bash 
# Test the initialize handshake
echo '{"jsonrpc":"2.0","id":1,"method":"initialize","params":{"protocolVersion":"2024-11-05","capabilities":{},"clientInfo":{"name":"test","version":"1.0"}}}' | python stdio_server.py

# Test listing tools (need to send initialize first)
(
echo '{"jsonrpc":"2.0","id":1,"method":"initialize","params":{"protocolVersion":"2024-11-05","capabilities":{},"clientInfo":{"name":"test","version":"1.0"}}}'
echo '{"jsonrpc":"2.0","method":"notifications/initialized"}'
echo '{"jsonrpc":"2.0","id":2,"method":"tools/list"}'
) | python stdio_server.py

# Test calling a tool
(
echo '{"jsonrpc":"2.0","id":1,"method":"initialize","params":{"protocolVersion":"2024-11-05","capabilities":{},"clientInfo":{"name":"test","version":"1.0"}}}'
echo '{"jsonrpc":"2.0","method":"notifications/initialized"}'
echo '{"jsonrpc":"2.0","id":2,"method":"tools/call","params":{"name":"echo","arguments":{"message":"Hello, MCP!"}}}'
) | python stdio_server.py

Key Implementation Details

1. Clean Stream Separation

python 
# Logging goes to stderr ONLY
logging.basicConfig(handlers=[logging.StreamHandler(sys.stderr)])

# Responses go to stdout ONLY
sys.stdout.write(json_str + "\n")
sys.stdout.flush()

2. Always Flush Output

python 
# Every write to stdout must be flushed
sys.stdout.write(json_str + "\n")
sys.stdout.flush()  # Never forget this!

3. Graceful EOF Handling

python 
line = sys.stdin.readline()
if not line:  # Empty string = EOF
    logger.info("EOF received, shutting down")
    break

4. Error Recovery

python 
# Don't crash on bad JSON - log and continue
try:
    message = json.loads(line)
except json.JSONDecodeError as e:
    logger.error(f"JSON parse error: {e}")
    continue  # Keep processing
Common Mistakes to Avoid
  • Using print() anywhere (it goes to stdout!)
  • Forgetting to call flush() after writing
  • Pretty-printing JSON responses (breaks line-based protocol)
  • Not handling EOF properly (server hangs)
  • Importing libraries that print to stdout
