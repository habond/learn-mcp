Error Handling

Master JSON-RPC error codes and build robust error responses.

Error Response Structure

When a request fails, the server returns an error response instead of a result. The error object has three fields:

json 
{
    "jsonrpc": "2.0",
    "id": 1,
    "error": {
        "code": -32601,           // Required: numeric error code
        "message": "Method not found",  // Required: short description
        "data": {                 // Optional: additional error details
            "method": "unknown/method",
            "suggestion": "Did you mean 'tools/list'?"
        }
    }
}
Never Both!

A response must have either result OR error, never both. If you include both, clients may behave unpredictably.

Standard Error Codes

JSON-RPC 2.0 defines these standard error codes. The range -32000 to -32099 is reserved for implementation-defined server errors.

Code Name Meaning
-32700 Parse error Invalid JSON received
-32600 Invalid Request JSON is not a valid request object
-32601 Method not found Method does not exist
-32602 Invalid params Invalid method parameters
-32603 Internal error Internal server error
-32000 to -32099 Server error Reserved for implementation errors

Implementing Error Codes

Define error codes as constants for consistency across your codebase:

python 
from enum import IntEnum
from typing import Any, Optional


class JsonRpcErrorCode(IntEnum):
    """Standard JSON-RPC 2.0 error codes."""

    # Standard errors
    PARSE_ERROR = -32700
    INVALID_REQUEST = -32600
    METHOD_NOT_FOUND = -32601
    INVALID_PARAMS = -32602
    INTERNAL_ERROR = -32603

    # Server errors (implementation-defined)
    SERVER_ERROR = -32000
    SERVER_NOT_INITIALIZED = -32002


# Human-readable messages for each code
ERROR_MESSAGES = {
    JsonRpcErrorCode.PARSE_ERROR: "Parse error",
    JsonRpcErrorCode.INVALID_REQUEST: "Invalid request",
    JsonRpcErrorCode.METHOD_NOT_FOUND: "Method not found",
    JsonRpcErrorCode.INVALID_PARAMS: "Invalid params",
    JsonRpcErrorCode.INTERNAL_ERROR: "Internal error",
    JsonRpcErrorCode.SERVER_ERROR: "Server error",
    JsonRpcErrorCode.SERVER_NOT_INITIALIZED: "Server not initialized",
}

Creating Error Responses

Build a helper function to create properly formatted error responses:

python 
def create_error_response(
    request_id: Optional[int],
    code: JsonRpcErrorCode,
    message: str = None,
    data: Any = None
) -> dict:
    """
    Create a JSON-RPC error response.

    Args:
        request_id: The ID from the original request (None if parse error)
        code: The error code from JsonRpcErrorCode
        message: Custom message (uses default if not provided)
        data: Optional additional error data

    Returns:
        A properly formatted JSON-RPC error response
    """
    error = {
        "code": int(code),
        "message": message or ERROR_MESSAGES.get(code, "Unknown error")
    }

    if data is not None:
        error["data"] = data

    return {
        "jsonrpc": "2.0",
        "id": request_id,  # Can be None for parse errors
        "error": error
    }


# Examples of creating error responses

# Method not found
error1 = create_error_response(
    request_id=1,
    code=JsonRpcErrorCode.METHOD_NOT_FOUND,
    data={"method": "foo/bar"}
)
# {"jsonrpc": "2.0", "id": 1, "error": {"code": -32601, "message": "Method not found", "data": {"method": "foo/bar"}}}

# Invalid params with custom message
error2 = create_error_response(
    request_id=2,
    code=JsonRpcErrorCode.INVALID_PARAMS,
    message="Missing required parameter: 'name'",
    data={"required": ["name"], "received": []}
)

# Parse error (no request ID available)
error3 = create_error_response(
    request_id=None,  # Can't know the ID if JSON was invalid
    code=JsonRpcErrorCode.PARSE_ERROR,
    message="Unexpected token at position 42"
)

Exception-Based Error Handling

Use custom exceptions to propagate errors through your code cleanly:

python 
class JsonRpcError(Exception):
    """Base exception for JSON-RPC errors."""

    def __init__(
        self,
        code: JsonRpcErrorCode,
        message: str = None,
        data: Any = None
    ):
        self.code = code
        self.message = message or ERROR_MESSAGES.get(code, "Unknown error")
        self.data = data
        super().__init__(self.message)

    def to_response(self, request_id: Optional[int]) -> dict:
        """Convert exception to JSON-RPC error response."""
        return create_error_response(
            request_id=request_id,
            code=self.code,
            message=self.message,
            data=self.data
        )


class MethodNotFoundError(JsonRpcError):
    """Raised when the requested method doesn't exist."""

    def __init__(self, method: str):
        super().__init__(
            code=JsonRpcErrorCode.METHOD_NOT_FOUND,
            message=f"Method not found: {method}",
            data={"method": method}
        )


class InvalidParamsError(JsonRpcError):
    """Raised when method parameters are invalid."""

    def __init__(self, message: str, details: dict = None):
        super().__init__(
            code=JsonRpcErrorCode.INVALID_PARAMS,
            message=message,
            data=details
        )


class InternalError(JsonRpcError):
    """Raised for unexpected server errors."""

    def __init__(self, message: str = "An internal error occurred"):
        super().__init__(
            code=JsonRpcErrorCode.INTERNAL_ERROR,
            message=message
        )

Error Handling in Practice

Here's how to use these exceptions in a request handler:

python 
import json
import traceback


def handle_request(raw_json: str) -> dict:
    """
    Process a JSON-RPC request and return a response.

    This function handles all error cases gracefully.
    """
    request_id = None

    try:
        # Step 1: Parse JSON
        try:
            request = json.loads(raw_json)
        except json.JSONDecodeError as e:
            raise JsonRpcError(
                code=JsonRpcErrorCode.PARSE_ERROR,
                message=f"Invalid JSON: {e}"
            )

        # Step 2: Validate request structure
        if not isinstance(request, dict):
            raise JsonRpcError(
                code=JsonRpcErrorCode.INVALID_REQUEST,
                message="Request must be a JSON object"
            )

        request_id = request.get("id")

        if request.get("jsonrpc") != "2.0":
            raise JsonRpcError(
                code=JsonRpcErrorCode.INVALID_REQUEST,
                message="jsonrpc must be '2.0'"
            )

        method = request.get("method")
        if not method or not isinstance(method, str):
            raise JsonRpcError(
                code=JsonRpcErrorCode.INVALID_REQUEST,
                message="Missing or invalid 'method' field"
            )

        # Step 3: Route to handler
        result = dispatch_method(method, request.get("params", {}))

        # Step 4: Return success response
        return {
            "jsonrpc": "2.0",
            "id": request_id,
            "result": result
        }

    except JsonRpcError as e:
        # Known error - return proper error response
        return e.to_response(request_id)

    except Exception as e:
        # Unexpected error - log and return internal error
        print(f"Unexpected error: {traceback.format_exc()}")
        return create_error_response(
            request_id=request_id,
            code=JsonRpcErrorCode.INTERNAL_ERROR,
            message="An unexpected error occurred"
        )


def dispatch_method(method: str, params: dict) -> dict:
    """Dispatch to the appropriate method handler."""
    handlers = {
        "tools/list": handle_tools_list,
        "tools/call": handle_tools_call,
    }

    if method not in handlers:
        raise MethodNotFoundError(method)

    return handlers[method](params)


def handle_tools_list(params: dict) -> dict:
    """Handle tools/list method."""
    return {"tools": []}


def handle_tools_call(params: dict) -> dict:
    """Handle tools/call method."""
    if "name" not in params:
        raise InvalidParamsError(
            "Missing required parameter: 'name'",
            {"required": ["name"], "received": list(params.keys())}
        )
    return {"content": [{"type": "text", "text": "Tool result"}]}
Security Tip

Never expose internal error details (stack traces, file paths, database queries) to clients. Log them server-side for debugging, but return generic messages to clients.

Testing Error Responses

Test each error path to ensure robust error handling:

python 
# Test parse error
result = handle_request("not valid json{{{")
assert result["error"]["code"] == -32700

# Test invalid request
result = handle_request('{"jsonrpc": "1.0", "method": "test"}')
assert result["error"]["code"] == -32600

# Test method not found
result = handle_request('{"jsonrpc": "2.0", "id": 1, "method": "unknown"}')
assert result["error"]["code"] == -32601
assert result["id"] == 1  # ID should be preserved

# Test invalid params
result = handle_request('{"jsonrpc": "2.0", "id": 2, "method": "tools/call", "params": {}}')
assert result["error"]["code"] == -32602

print("All error handling tests passed!")
